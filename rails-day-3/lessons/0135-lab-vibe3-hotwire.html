🍽 เทียบให้เห็นภาพ — อ่านสูตร vs ชิมจริง

พ่อครัวยื่นสูตรมาบอกว่า "ใส่น้ำปลาแล้ว ใส่มะนาวแล้ว" — อ่านสูตรแล้วดูครบ แต่ยังไม่รู้ว่าอร่อยไหม · เจ้าของร้านที่ดีตักชิมเอง และให้คนสองโต๊ะสั่งพร้อมกัน ดูว่าเสียงตามสายดังถึงทั้งสองโต๊ะจริงไหม · อ่านโค้ด = อ่านสูตร · เปิดเบราว์เซอร์สองหน้าต่างลองเอง = ชิม

ที่ไม่เหมือนกัน: การชิมอาหารบอกได้ทุกอย่างที่ลูกค้าจะเจอ · แต่การลองในเบราว์เซอร์บอกไม่ได้ทุกอย่าง — บางบั๊กไม่มีอาการให้เห็นด้วยตา (เช่น ลืม disconnect ในขั้น 3) จึงต้องมีตรวจด้วยคำสั่งควบคู่กันเสมอ

ข้อ 5 แบบเห็นกับตา: ใส่ console.log ชั่วคราว

grep บอกได้แค่ว่า "มีคำว่า disconnect ในไฟล์" · อยากเห็นว่ามันถูกเรียกจริง ให้เปิด app/javascript/controllers/character_counter_controller.js แล้วเติม 2 บรรทัดที่มีป้าย // ← เติมชั่วคราว (ที่เหลือคือไฟล์เดิมจากบท 134 ไม่ต้องแตะ): 

import { Controller } from "@hotwired/stimulus"

// ตัวนับอักษรของช่องพิมพ์คอมเมนต์ (อยู่ใน comments/_form.html.erb) แสดงเป็น "12 / 1000"
export default class extends Controller {
  static targets = ["input", "output"]

  // ฟอร์มโผล่ (โหลดหน้า หรือ Turbo Stream ใส่ฟอร์มใหม่): เริ่มฟังการพิมพ์
  connect() {
    console.log("ตัวนับ: connect")        // ← เติมชั่วคราว
    this.onInput = this.update.bind(this)
    this.inputTarget.addEventListener("input", this.onInput)
    this.update()
  }

  // ฟอร์มหายไป: ถอด listener ที่ connect ผูกไว้
  disconnect() {
    console.log("ตัวนับ: disconnect")     // ← เติมชั่วคราว
    this.inputTarget.removeEventListener("input", this.onInput)
  }

  update() {
    this.outputTarget.textContent = `${this.inputTarget.value.length} / ${this.inputTarget.maxLength}`
  }
}

ไม่เจอ · ผลจริง (ลองทั้ง zsh ที่เป็นค่าตั้งต้นของ terminal บน Mac และ bash ได้เหมือนกัน):

  • terminal (ตัวที่อ่านคำสั่งเรียกว่า shell) แปลง ** เป็นรายชื่อไฟล์ก่อนส่งให้ grep · ** ที่ไม่มีอะไรต่อท้าย มันอ่านเหมือน * ธรรมดา = "ของที่อยู่ชั้นแรกของโฟลเดอร์" → ได้แค่ app/javascript/application.js กับโฟลเดอร์ controllers
  • grep ไม่มี -r จึงไม่เข้าไปในโฟลเดอร์ บ่นว่า "Is a directory" แล้วข้ามไป · ไฟล์ตัวนับอยู่ในโฟลเดอร์นั้น จึงไม่ถูกค้นเลย
  • อันตรายตรงที่ข้อความนี้ดูเหมือน "ไม่เจอ disconnect" — คุณอาจสรุปผิดว่างานไม่ผ่าน (หรือแย่กว่า: ใช้คำสั่งนี้ตรวจงานที่ไม่มี disconnect จริง แล้วไม่ทันสังเกตว่ามันไม่ได้ค้นอะไรเลย)

① เติม 3 บรรทัดนี้ท้าย app/views/posts/show.html.erb ชั่วคราว (แบบที่ AI ชอบเขียน):

<script>

glob · * · **
