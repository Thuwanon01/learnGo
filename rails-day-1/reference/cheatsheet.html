class GreetingsController < ApplicationController      # ⚠ < ต้องอยู่บรรทัดเดียวกับ class
  before_action :reject_long_name, only: %i[index]       # รันก่อน action

  def index
    @name = params[:name].presence || "PEA"             # @ = สะพานไป view · ไม่มี render = implicit render
  end

  def new                                                  # ว่าง → render new.html.erb (ฟอร์ม)
  end

  def create
    @name = params[:name]
    if @name.present?
      redirect_to hello_path(name: @name), notice: "สร้างแล้ว"   # สำเร็จ → redirect + flash
    else
      render :new, status: :unprocessable_content            # ไม่ผ่าน → render ฟอร์มเดิม (422)
    end
  end

  private                                                  # ข้างล่างไม่ใช่ action · route ชี้มาไม่ได้

  def reject_long_name
    if params[:name].to_s.length > 20                       # .to_s กัน nil.length
      redirect_to root_path, alert: "ชื่อยาวเกินไป"          # redirect ใน before_action = action ไม่ถูกรัน
    end
  end
end
