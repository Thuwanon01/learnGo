2 · REST — method กับความหมาย

Methodแปลว่ามี body ไหมidempotentตัวอย่าง
GETอ่าน — ห้ามเปลี่ยนอะไรไม่✔GET /recipes/12
POSTสร้างใหม่ / สั่งให้ทำใช่✘POST /recipes
PUTแทนที่ทั้งก้อนใช่✔PUT /recipes/12
PATCHแก้บางส่วนใช่✘*PATCH /recipes/12
DELETEลบไม่✔DELETE /recipes/12

idempotent = ยิงซ้ำกี่ครั้งผลสุดท้ายเหมือนยิงครั้งเดียว — ไม่ได้แปลว่า “ได้ status เดิม” · DELETE ครั้งแรกได้ 204 ครั้งที่สองได้ 404 ก็ยังถือว่า idempotent เพราะสถานะของ server เหมือนกัน

* PATCH idempotent หรือไม่ขึ้นกับว่าเขียนยังไง — {"stock": 5} idempotent แต่ {"stock": "+1"} ไม่

PUT ที่คนใหม่พลาดที่สุด — PUT /users/1 ด้วย body {"name":"ใหม่"} ตามนิยามคือ แทนที่ทั้ง resource แปลว่า email ที่ไม่ได้ส่งมาควรถูกลบทิ้ง ถ้าอยากแก้แค่ชื่อต้องใช้ PATCH

3 · Status code — เลือกตัวไหน

Codeใช้เมื่ออย่าใช้เมื่อ
200 OKสำเร็จ และมี body จะตอบสร้างของใหม่ (ใช้ 201)
201 Createdสร้าง resource สำเร็จ — ควรใส่ Location ชี้ของใหม่แก้ของเดิม
204 No Contentสำเร็จ แต่ไม่มีอะไรจะตอบ เช่น DELETEเมื่อยังอยากส่ง body (204 ห้ามมี body)
400 Bad Requestรูปแบบ request ผิด / validation ไม่ผ่านยังไม่ได้ล็อกอิน (ใช้ 401)
401 Unauthorizedยังไม่รู้ว่าคุณเป็นใคร — ไม่มี token / token ผิดรู้ว่าเป็นใครแล้วแต่ห้ามทำ (ใช้ 403)
403 Forbiddenรู้ว่าคุณเป็นใครแล้ว แต่ไม่มีสิทธิ์token หมดอายุ (ใช้ 401)
404 Not Foundไม่มี resource นี้มีของแต่ห้ามดู (403 หรือ 404 ก็ได้ ถ้าตั้งใจซ่อน)
409 Conflictชนกับสถานะปัจจุบัน เช่น email ซ้ำข้อมูลผิดรูป (ใช้ 400)
500 Internal Server Errorเราผิดเอง — bug, DB ล่มผู้ใช้ส่งข้อมูลผิด (นั่นคือ 4xx)
เส้นแบ่งเดียวที่ต้องจำ

4xx = ฝั่ง client ต้องแก้ · 5xx = ฝั่งเราต้องแก้
ตอบ 200 พร้อม {"error": "..."} คือความผิดพลาดที่แพงที่สุด — client library ที่เช็ค status จะคิดว่าสำเร็จ และ dashboard จะมองไม่เห็น error rate เลย

6 · net/http — โครงที่ก็อปไปใช้ได้

package main

import (
    "encoding/json"
    "log"
    "net/http"
)

type User struct {
    ID   string `json:"id"`
    Name string `json:"name"`
}

func getUser(w http.ResponseWriter, r *http.Request) {
    id := r.PathValue("id")          // Go 1.22+

    // ลำดับนี้ห้ามสลับ: Header → WriteHeader → Encode
    w.Header().Set("Content-Type", "application/json; charset=utf-8")
    w.WriteHeader(http.StatusOK)
    json.NewEncoder(w).Encode(User{ID: id, Name: "Somchai"})
}

func main() {
    mux := http.NewServeMux()
    mux.HandleFunc("GET /health", func(w http.ResponseWriter, r *http.Request) {
        w.Write([]byte("ok"))
    })
    mux.HandleFunc("GET /users/{id}", getUser)

    log.Fatal(http.ListenAndServe(":8080", mux))
}

ทำไม w ไม่ใช่ pointer แต่ r เป็น — http.ResponseWriter เป็น interface (ข้างในมี pointer อยู่แล้ว) ส่วน http.Request เป็น struct ก้อนใหญ่ จึงส่งเป็น *http.Request

สิ่งที่ net/http ทำให้ฟรี

# ยิง POST ไปที่ route ที่ลงเป็น GET

Allow: GET, HEAD
Content-Type: text/plain; charset=utf-8

7 · net/http — ไวยากรณ์ pattern (Go 1.22+)

เขียนแมตช์อะไรหมายเหตุ
"/hello"path เป๊ะ ทุก method
"GET /hello"เฉพาะ GET (และ HEAD)method ต้องตัวใหญ่ + เว้นวรรค 1 ที
"/users/{id}"1 segmentอ่านด้วย r.PathValue("id")
"/files/{path...}"ที่เหลือทั้งหมดต้องอยู่ท้าย pattern
"/files/"subtree — ทุกอย่างใต้ /files/ยิง /files จะถูก redirect 307
"/exact/{$}"/exact/ เป๊ะ ๆ ห้าม subtree

ลำดับความสำคัญ: pattern ที่เจาะจงกว่าชนะเสมอ ไม่เกี่ยวกับลำดับที่ลงทะเบียน — /users/me ชนะ /users/{id} · และ ServeMux ไม่ panic เมื่อลงทั้งสองอัน

ไวยากรณ์ที่พิมพ์ผิดแล้วพังเงียบ ๆ

pattern="get /hello"    // ลงทะเบียนผ่าน! แต่ GET จริงได้ 405 Allow: get
pattern="GET/hello"     // Go ตีเป็น host="GET" path="/hello" → 404 ทุกครั้ง เงียบสนิท
pattern="GET  /hello"   // เว้นวรรคเกิน → ใช้ได้ปกติ
pattern="GET hello"     // PANIC: parsing "GET hello": at offset 4: host/path missing /

8 · Gin — โครงที่ก็อปไปใช้ได้

package main

import (
    "net/http"
    "github.com/gin-gonic/gin"
)

func NewRouter() *gin.Engine {
    r := gin.Default()                // = New() + Logger() + Recovery()
    r.HandleMethodNotAllowed = true    // อยากได้ 405 ต้องเปิดเอง

    v1 := r.Group("/api/v1")
    {
        v1.GET("/users/:id", getUser)
        v1.POST("/users", createUser)
    }
    return r
}

func getUser(c *gin.Context) {
    c.JSON(http.StatusOK, gin.H{"id": c.Param("id")})
}

func main() { NewRouter().Run(":8080") }

แยก NewRouter() ออกจาก main() ไม่ใช่เรื่องความสวย — เป็นเงื่อนไขที่ทำให้เขียนเทสด้วย httptest ได้ (หัวข้อ 12) เพราะเทสต้องหยิบ *gin.Engine มาใช้โดยไม่เปิด port

10 · Gin — binding tag + ข้อความ error จริง

type CreateRecipeRequest struct {
    Name       string  `json:"name"       binding:"required"`
    Email      string  `json:"email"      binding:"required,email"`
    Level      int     `json:"level"      binding:"min=1,max=5"`
    Difficulty string  `json:"difficulty" binding:"oneof=easy medium hard"`
    Note       *string `json:"note"`   // pointer = แยก "ไม่ส่งมา" ออกจาก "ส่งค่าว่าง"
}

func create(c *gin.Context) {
    var req CreateRecipeRequest
    if err := c.ShouldBindJSON(&req); err != nil {   // ShouldBind เสมอ ไม่ใช่ Bind
        c.JSON(http.StatusBadRequest, gin.H{"message": err.Error()})
        return
    }
    c.JSON(http.StatusCreated, req)
}

ข้อความ error ที่ได้จริง

body={"age":0}              → Key: 'ReqAge.Age' Error:Field validation for 'Age' failed on the 'required' tag
body={"email":"not-an-email"} → Key: 'ReqEmail.Email' Error:Field validation for 'Email' failed on the 'email' tag
body={"level":9,...}        → Key: 'ReqRange.Level' Error:Field validation for 'Level' failed on the 'max' tag
                              Key: 'ReqRange.Difficulty' Error:Field validation for 'Difficulty' failed on the 'oneof' tag

ผิดหลาย field พร้อมกัน → error ทุกอันต่อกันด้วย \n ในสตริงเดียว

กับดัก required + zero value

Age int `binding:"required"` แล้วผู้ใช้ส่ง {"age":0} มา → ไม่ผ่าน เพราะ validator เช็ค “ไม่ใช่ zero value” ไม่ใช่ “มี key นี้” เจอกับ 0, "", false เหมือนกันหมด
แก้ด้วย pointer: Age *int `binding:"required"` — ส่ง 0 มาผ่าน แต่ไม่ส่งมาเลยไม่ผ่าน

เมธอดอ่านจากtag ที่ใช้
ShouldBindJSONbodyjson + binding
ShouldBindQueryquery stringform + binding
ShouldBindUripath paramuri + binding

11 · Gin — middleware & CORS

func RequireAuth() gin.HandlerFunc {
    return func(c *gin.Context) {
        if c.GetHeader("Authorization") == "" {
            c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"message": "no token"})
            return
        }
        c.Set("userID", "u-123")
        c.Next()                 // ปล่อยให้ตัวถัดไปทำงาน
        // โค้ดตรงนี้รัน "ขากลับ" หลัง handler เสร็จแล้ว
    }
}

r.Use(RequireAuth())              // ทั้งเว็บ
admin := r.Group("/admin")
admin.Use(RequireAuth())          // เฉพาะกลุ่มนี้
เขียนผล
c.Next()เรียกตัวถัดไป แล้วกลับมาทำโค้ดที่เหลือ
c.Abort() เปล่า ๆหยุด handler ถัดไป แต่ไม่ได้เขียน response → client ได้ 200 body ว่าง
c.AbortWithStatusJSON(401, …)หยุด + ตอบ JSON — ใช้ตัวนี้เสมอ
c.Set / c.Getส่งค่าให้ handler ถัดไป · c.MustGet panic ถ้าไม่มี key

CORS — ผลรันจริง

# request จริง (มี Origin) กับ cors.Default()
$ curl -s -i -H "Origin: https://evil.example.com" http://127.0.0.1:39182/data
HTTP/1.1 200 OK
Access-Control-Allow-Origin: *

# preflight
$ curl -s -i -X OPTIONS -H "Origin: https://evil.example.com" \
    -H "Access-Control-Request-Method: POST" http://127.0.0.1:39182/data
HTTP/1.1 204 No Content
Access-Control-Allow-Headers: Origin,Content-Length,Content-Type
Access-Control-Allow-Methods: GET,POST,PUT,PATCH,DELETE,HEAD,OPTIONS
Access-Control-Allow-Origin: *
Access-Control-Max-Age: 43200
⚠ cors.Default() ไม่อนุญาต Authorization

ดู Access-Control-Allow-Headers ข้างบน — มีแค่ Origin, Content-Length, Content-Type · เว็บที่ส่ง JWT ผ่าน Authorization: Bearer … จะโดนเบราว์เซอร์บล็อกทั้งที่ server ตอบ 204
ต้องเขียนเองด้วย cors.New:

r.Use(cors.New(cors.Config{
    AllowOrigins: []string{"https://app.example.com"},
    AllowMethods: []string{"GET", "POST", "PUT", "DELETE"},
    AllowHeaders: []string{"Origin", "Content-Type", "Authorization"},
}))

CORS ไม่ใช่ระบบความปลอดภัยของ server — เป็นกฎที่เบราว์เซอร์บังคับตัวเอง curl และ Postman ไม่สนใจ CORS เลย ยิงได้ตลอด

12 · httptest — โครงเทสที่ก็อปไปใช้ได้

func TestGetUser(t *testing.T) {
    gin.SetMode(gin.TestMode)
    router := NewRouter()

    w := httptest.NewRecorder()
    req := httptest.NewRequest("GET", "/users/1", nil)
    router.ServeHTTP(w, req)

    assert.Equal(t, 200, w.Code)
    assert.JSONEq(t, `{"id":"1","name":"Somchai"}`, w.Body.String())
    assert.Equal(t, "application/json; charset=utf-8", w.Header().Get("Content-Type"))
}
เขียนได้
w.Codeint เช่น 200
w.Body.String()body ทั้งก้อนเป็นสตริง
w.Header().Get("X")header — ไม่ต้องเรียก w.Result()
httptest.NewRequestpanic ถ้า URL พัง — ใช้ในเทส
http.NewRequestคืน error — ใช้ในโค้ด production
assert.JSONEqเทียบเชิงความหมาย — สลับลำดับ key ก็ผ่าน
assert.Equal กับสตริงเทียบตัวอักษร — เปราะ ใช้เฉพาะตอนอยากล็อกรูปแบบเป๊ะ

NewRecorder = ยิงเข้า router ในหน่วยความจำ ไม่เปิด port — ใช้ 99% ของเวลา  |  httptest.NewServer = เปิด server จริงบน port สุ่ม ใช้เมื่อต้องทดสอบตัว HTTP client ของเราเอง

14 · Redis — สองคำสั่งที่เพิ่มจาก Day 7

เรื่อง Redis ที่เหลือทั้งหมดอยู่ใน บทเรียน 32 และ cheat sheet Day 7 แล้ว สไลด์ Day 8 มีของใหม่จริง ๆ แค่สองคำสั่งนี้

# SET พร้อมตั้งเวลาหมดอายุในคำสั่งเดียว (นิยมใช้สุด)
SET username "devpool" EX 60

# GETDEL — อ่านค่าแล้วลบทิ้งในคำสั่งเดียว (atomic)
GETDEL username

16 · ตารางกับดักรวมทั้งวัน

เขียนแบบนี้สิ่งที่เกิดขึ้นจริงแก้
w.Header().Set(…) หลัง w.WriteHeader(…)ไม่มีผล เงียบสนิท ไม่มี error ไม่มี logลำดับ Header → WriteHeader → Write
w.WriteHeader(400) หลังเขียน body แล้วstatus ยังเป็น 200 · server log http: superfluous response.WriteHeader callreturn ทันทีหลังตอบ error
json.NewEncoder(w).Encode(…) โดยไม่ set Content-Typeclient ได้ text/plain; charset=utf-8set เองก่อน WriteHeader
mux.HandleFunc("GET/hello", …)Go ตีเป็น host=GET → 404 ทุก requestเว้นวรรคหลัง method
Gin: ยิง method ผิดได้ 404 ไม่ใช่ 405r.HandleMethodNotAllowed = true
c.BindJSON แล้วเขียน c.JSON(422, …) ต่อได้ 400 ไม่ใช่ 422 · [GIN-debug] [WARNING] Headers were already written.ใช้ c.ShouldBindJSON เสมอ
Age int `binding:"required"`ส่ง 0 มาแล้วไม่ผ่านเปลี่ยนเป็น *int
c.Abort() เปล่า ๆclient ได้ 200 body ว่างc.AbortWithStatusJSON(…)
r.Use(cors…) หลัง ลง route แล้วroute นั้นไม่ได้ middleware เลยUse ก่อนลง route เสมอ
cors.Default() + JWTเบราว์เซอร์บล็อก Authorizationcors.New แล้วใส่ AllowHeaders เอง
ทดสอบ CORS ด้วย curl ธรรมดาไม่มี CORS header ออกมาสักตัวต้องใส่ -H "Origin: …"
assert.Equal กับ body JSONพังเมื่อลำดับ key เปลี่ยนassert.JSONEq
เทียบ "application/json" เฉย ๆพัง — Gin ใส่ ; charset=utf-8 ด้วยเทียบสตริงเต็ม
router.GET("/swagger", …)/swagger/index.html → 404 page not foundต้องเป็น /swagger/*any
ลืม _ "wongnok/docs"หน้า UI ยัง 200 แต่ /swagger/doc.json คืน 500 body ว่าง → UI เปิดได้แต่ไม่มี endpointใส่ blank import
แก้ annotation แล้วไม่รัน swag initเอกสารเป็นของเก่ารันใหม่ทุกครั้ง + restart server
