2 · REST — method กับความหมาย

Methodแปลว่ามี body ไหมidempotentตัวอย่าง
GETอ่าน — ห้ามเปลี่ยนอะไรไม่✔GET /recipes/12
POSTสร้างใหม่ / สั่งให้ทำใช่✘POST /recipes
PUTแทนที่ทั้งก้อนใช่✔PUT /recipes/12
PATCHแก้บางส่วนใช่✘*PATCH /recipes/12
DELETEลบไม่✔DELETE /recipes/12

idempotent = ยิงซ้ำกี่ครั้งผลสุดท้ายเหมือนยิงครั้งเดียว — ไม่ได้แปลว่า “ได้ status เดิม” · DELETE ครั้งแรกได้ 204 ครั้งที่สองได้ 404 ก็ยังถือว่า idempotent เพราะสถานะของ server เหมือนกัน

* PATCH idempotent หรือไม่ขึ้นกับว่าเขียนยังไง — {"stock": 5} idempotent แต่ {"stock": "+1"} ไม่

PUT ที่คนใหม่พลาดที่สุด — PUT /users/1 ด้วย body {"name":"ใหม่"} ตามนิยามคือ แทนที่ทั้ง resource แปลว่า email ที่ไม่ได้ส่งมาควรถูกลบทิ้ง ถ้าอยากแก้แค่ชื่อต้องใช้ PATCH

3 · Status code — เลือกตัวไหน

Codeใช้เมื่ออย่าใช้เมื่อ
200 OKสำเร็จ และมี body จะตอบสร้างของใหม่ (ใช้ 201)
201 Createdสร้าง resource สำเร็จ — ควรใส่ Location ชี้ของใหม่แก้ของเดิม
204 No Contentสำเร็จ แต่ไม่มีอะไรจะตอบ เช่น DELETEเมื่อยังอยากส่ง body (204 ห้ามมี body)
400 Bad Requestรูปแบบ request ผิด / validation ไม่ผ่านยังไม่ได้ล็อกอิน (ใช้ 401)
401 Unauthorizedยังไม่รู้ว่าคุณเป็นใคร — ไม่มี token / token ผิดรู้ว่าเป็นใครแล้วแต่ห้ามทำ (ใช้ 403)
403 Forbiddenรู้ว่าคุณเป็นใครแล้ว แต่ไม่มีสิทธิ์token หมดอายุ (ใช้ 401)
404 Not Foundไม่มี resource นี้มีของแต่ห้ามดู (403 หรือ 404 ก็ได้ ถ้าตั้งใจซ่อน)
409 Conflictชนกับสถานะปัจจุบัน เช่น email ซ้ำข้อมูลผิดรูป (ใช้ 400)
500 Internal Server Errorเราผิดเอง — bug, DB ล่มผู้ใช้ส่งข้อมูลผิด (นั่นคือ 4xx)
เส้นแบ่งเดียวที่ต้องจำ

4xx = ฝั่ง client ต้องแก้ · 5xx = ฝั่งเราต้องแก้
ตอบ 200 พร้อม {"error": "..."} คือความผิดพลาดที่แพงที่สุด — client library ที่เช็ค status จะคิดว่าสำเร็จ และ dashboard จะมองไม่เห็น error rate เลย

6 · net/http — โครงที่ก็อปไปใช้ได้

package main

import (
    "encoding/json"
    "log"
    "net/http"
)

type User struct {
    ID   string `json:"id"`
    Name string `json:"name"`
}

func getUser(w http.ResponseWriter, r *http.Request) {
    id := r.PathValue("id")          // Go 1.22+

    // ลำดับนี้ห้ามสลับ: Header → WriteHeader → Encode
    w.Header().Set("Content-Type", "application/json; charset=utf-8")
    w.WriteHeader(http.StatusOK)
    json.NewEncoder(w).Encode(User{ID: id, Name: "Somchai"})
}

func main() {
    mux := http.NewServeMux()
    mux.HandleFunc("GET /health", func(w http.ResponseWriter, r *http.Request) {
        w.Write([]byte("ok"))
    })
    mux.HandleFunc("GET /users/{id}", getUser)

    log.Fatal(http.ListenAndServe(":8080", mux))
}

ทำไม w ไม่ใช่ pointer แต่ r เป็น — http.ResponseWriter เป็น interface (ข้างในมี pointer อยู่แล้ว) ส่วน http.Request เป็น struct ก้อนใหญ่ จึงส่งเป็น *http.Request

7 · net/http — ไวยากรณ์ pattern (Go 1.22+)

เขียนแมตช์อะไรหมายเหตุ
"/hello"path เป๊ะ ทุก method
"GET /hello"เฉพาะ GET (และ HEAD)method ต้องตัวใหญ่ + เว้นวรรค 1 ที
"/users/{id}"1 segmentอ่านด้วย r.PathValue("id")
"/files/{path...}"ที่เหลือทั้งหมดต้องอยู่ท้าย pattern
"/files/"subtree — ทุกอย่างใต้ /files/ยิง /files จะถูก redirect 307
"/exact/{$}"/exact/ เป๊ะ ๆ ห้าม subtree

ลำดับความสำคัญ: pattern ที่เจาะจงกว่าชนะเสมอ ไม่เกี่ยวกับลำดับที่ลงทะเบียน — /users/me ชนะ /users/{id} · และ ServeMux ไม่ panic เมื่อลงทั้งสองอัน

ไวยากรณ์ที่พิมพ์ผิดแล้วพังเงียบ ๆ

pattern="get /hello"    // ลงทะเบียนผ่าน! แต่ GET จริงได้ 405 Allow: get
pattern="GET/hello"     // Go ตีเป็น host="GET" path="/hello" → 404 ทุกครั้ง เงียบสนิท
pattern="GET  /hello"   // เว้นวรรคเกิน → ใช้ได้ปกติ
pattern="GET hello"     // PANIC: parsing "GET hello": at offset 4: host/path missing /

8 · Gin — โครงที่ก็อปไปใช้ได้

package main

import (
    "net/http"
    "github.com/gin-gonic/gin"
)

func NewRouter() *gin.Engine {
    r := gin.Default()                // = New() + Logger() + Recovery()
    r.HandleMethodNotAllowed = true    // อยากได้ 405 ต้องเปิดเอง

    v1 := r.Group("/api/v1")
    {
        v1.GET("/users/:id", getUser)
        v1.POST("/users", createUser)
    }
    return r
}

func getUser(c *gin.Context) {
    c.JSON(http.StatusOK, gin.H{"id": c.Param("id")})
}

func main() { NewRouter().Run(":8080") }

แยก NewRouter() ออกจาก main() ไม่ใช่เรื่องความสวย — เป็นเงื่อนไขที่ทำให้เขียนเทสด้วย httptest ได้ (หัวข้อ 12) เพราะเทสต้องหยิบ *gin.Engine มาใช้โดยไม่เปิด port

10 · Gin — binding tag + ข้อความ error จริง

type CreateRecipeRequest struct {
    Name       string  `json:"name"       binding:"required"`
    Email      string  `json:"email"      binding:"required,email"`
    Level      int     `json:"level"      binding:"min=1,max=5"`
    Difficulty string  `json:"difficulty" binding:"oneof=easy medium hard"`
    Note       *string `json:"note"`   // pointer = แยก "ไม่ส่งมา" ออกจาก "ส่งค่าว่าง"
}

func create(c *gin.Context) {
    var req CreateRecipeRequest
    if err := c.ShouldBindJSON(&req); err != nil {   // ShouldBind เสมอ ไม่ใช่ Bind
        c.JSON(http.StatusBadRequest, gin.H{"message": err.Error()})
        return
    }
    c.JSON(http.StatusCreated, req)
}

11 · Gin — middleware & CORS

func RequireAuth() gin.HandlerFunc {
    return func(c *gin.Context) {
        if c.GetHeader("Authorization") == "" {
            c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"message": "no token"})
            return
        }
        c.Set("userID", "u-123")
        c.Next()                 // ปล่อยให้ตัวถัดไปทำงาน
        // โค้ดตรงนี้รัน "ขากลับ" หลัง handler เสร็จแล้ว
    }
}

r.Use(RequireAuth())              // ทั้งเว็บ
admin := r.Group("/admin")
admin.Use(RequireAuth())          // เฉพาะกลุ่มนี้
เขียนผล
c.Next()เรียกตัวถัดไป แล้วกลับมาทำโค้ดที่เหลือ
c.Abort() เปล่า ๆหยุด handler ถัดไป แต่ไม่ได้เขียน response → client ได้ 200 body ว่าง
c.AbortWithStatusJSON(401, …)หยุด + ตอบ JSON — ใช้ตัวนี้เสมอ
c.Set / c.Getส่งค่าให้ handler ถัดไป · c.MustGet panic ถ้าไม่มี key

CORS — ผลรันจริง

preflight = request OPTIONS ที่เบราว์เซอร์ยิงเองก่อน โดยที่โค้ดเราไม่ได้สั่ง เพื่อถามเซิร์ฟเวอร์ว่า “ขอส่ง method/header ชุดนี้ได้ไหม” ถ้าคำตอบไม่ผ่าน เบราว์เซอร์จะไม่ยิง request จริงเลย (ละเอียดใน บทเรียน 39)

Access-Control-Allow-Origin: *

# preflight
$ curl -s -i -X OPTIONS -H "Origin: https://evil.example.com" \

func mustEnv(key string) string {
    v, ok := os.LookupEnv(key)
    if !ok {
        log.Fatalf("missing required env %s", key)   // ตายตั้งแต่ตอน start ดีกว่าพังตอนมีคนใช้
    }
    return v
}

func envInt(key string, def int) int {
    raw, ok := os.LookupEnv(key)
    if !ok || raw == "" {
        return def
    }
    n, err := strconv.Atoi(strings.TrimSpace(raw))   // TrimSpace: Atoi ไม่ตัดช่องว่างให้
    if err != nil {
        log.Fatalf("env %s=%q is not an int: %v", key, raw, err)
    }
    return n
}

compose และ k8s — ไวยากรณ์อ้างอิง

สองบล็อกนี้เป็นไวยากรณ์สำหรับก็อปไปใช้ ไม่ใช่ผลรัน — ต่างจากบล็อกอื่นในหน้านี้

# docker-compose.yml
services:
  api:
    image: myapi:latest
    environment:
      LOG_LEVEL: INFO
      APP_PORT: "8080"
      POSTGRES_DSN: ${POSTGRES_DSN}   # หยิบจาก shell / .env ของ host
    env_file:
      - app.env
# k8s — ส่วน env ของ container
env:
  - name: LOG_LEVEL
    value: "INFO"

  - name: APP_PORT              # ค่าธรรมดา → ConfigMap
    valueFrom:
      configMapKeyRef:
        name: app-config
        key: APP_PORT

  - name: POSTGRES_DSN          # ความลับ → Secret
    valueFrom:
      secretKeyRef:
        name: app-secret
        key: POSTGRES_DSN

เส้นแบ่ง ConfigMap = ค่าที่เปิดให้ใครดูก็ได้ (log level, port, feature flag) · Secret = ค่าที่หลุดแล้วเดือดร้อน (DSN ที่มีรหัสผ่าน, API key, JWT secret) · ทั้งคู่จบลงที่ตัวแปร env ตัวเดียวกันในสายตาโปรแกรม — โค้ด Go เขียน os.LookupEnv("POSTGRES_DSN") เหมือนกันหมด ไม่ต้องรู้ว่ามาจากไหน

19 · ตารางแปลง API ที่ออกแบบแย่ → RESTful

ของเดิมที่ควรเป็นผิดกฎข้อไหน
GET /getUser?id=123 GET /users/123 ใช้กริยาใน path (กฎ 1) + ใช้ query แทน path param สำหรับระบุตัวตน (หัวข้อ 5)
POST /deleteUser?id=123 DELETE /users/123 ใช้ method ผิด ทำให้ไม่ idempotent (หัวข้อ 2) + action อยู่ใน URI (กฎ 7)
POST /createUser?name=Alice&password=123456 POST /users + body ความลับอยู่ใน query string โผล่ใน log (หัวข้อ 5) + กริยาใน path (กฎ 1)
GET /getPaidOrders GET /orders?status=paid ยัดตัวกรองไว้ในชื่อ path แทน query (กฎ 6) + กริยาใน path (กฎ 1)

ทั้งสี่คู่ลงทะเบียนบน Gin v1.12.0 จริงแล้วยิงเทียบกัน — คู่ที่ 1 กับ 4 ให้ผลลัพธ์เหมือนกันเป๊ะ เส้นแบ่งจึงไม่ได้อยู่ที่ “ทำงานได้ไหม” แต่อยู่ที่สามอย่างข้างล่างนี้

① เส้นที่ 3 — password ใน query โผล่ใน access log จริง

$ grep -n '123456' gin-access.log

gin.Logger() เขียน path + RawQuery ลง log ทุกครั้ง — รหัสผ่านจึงถูกบันทึกเป็น plaintext อ่านได้ตรง ๆ · ส่วน body ไม่ถูก log เลย ทั้งที่ server อ่านค่าได้ครบ · และนี่เพิ่งแค่ log ของ server เราเอง ยังไม่นับ proxy, CDN, browser history และ Referer ที่หัวข้อ 5 เตือนไว้

② เส้นที่ 2 — idempotency ไม่ได้ดูจาก status code

# ✔ DELETE /users/123 ยิงซ้ำ 3 ครั้ง
call 1 -> {"deleted":"123","remaining":1}                  | HTTP 200
call 2 -> {"error":"user not found","id":"123",…}          | HTTP 404
call 3 -> {"error":"user not found","id":"123",…}          | HTTP 404

# ✘ POST /deleteUser?id=123 ยิงซ้ำ 3 ครั้ง
call 1 -> {"deleted":"123","remaining":1}                  | HTTP 200
call 2 -> {"error":"user not found","id":"123",…}          | HTTP 404
call 3 -> {"error":"user not found","id":"123",…}          | HTTP 404
⚠ ยิงซ้ำแล้วไม่เห็นความต่าง — และนั่นคือประเด็น

status code ออกมา 200, 404, 404 เหมือนกันเป๊ะทั้งสองแบบ และ state ปลายทางก็เหมือนกัน · idempotency ไม่ใช่สิ่งที่ server สะท้อนออกมาเป็นตัวเลข แต่เป็นสัญญาระดับ HTTP spec (RFC 9110 กำหนดว่า DELETE idempotent ส่วน POST ไม่) ที่ client library, proxy และ retry layer เอาไปใช้ตัดสินใจว่า “ยิงซ้ำเองได้ไหมเมื่อ timeout” — ใช้ POST แปลว่าไม่มีใครกล้า retry ให้

อีกจุดหนึ่ง: 404 ที่ได้ตอนยิงซ้ำ หน้าตาแยกไม่ออกจาก 404 ที่เกิดจากลบของที่ไม่เคยมี — DELETE /users/999 ก็ได้ 404 ตั้งแต่ครั้งแรก · API หลายเจ้าจึงเลือกตอบ 204 No Content ทุกครั้งแทน เพื่อให้ผลลัพธ์เหมือนกันจริง ๆ ทุกครั้ง

③ เส้นที่ 1 และ 4 — ดึงค่าจากคนละที่

GET /getUser?id=123      -> {"extracted_by":"c.Query(\"id\")",  "id":"123", …}
GET /users/123           -> {"extracted_by":"c.Param(\"id\")",  "id":"123", …}
GET /orders?status=paid  -> {"extracted_by":"c.Query(\"status\")","filter_status":"paid", …}
POST /users (JSON body)  -> {"extracted_by":"c.ShouldBindJSON (request body)","name":"Alice","password_len":6}
กับดักคู่กับ ?status=paid 
$ curl -s -w ' | HTTP %{http_code}\n' 'http://127.0.0.1:18080/orders'
{"filter_status":"","orders":[{"id":"o1","status":"paid"},{"id":"o2","status":"pending"},
 {"id":"o3","status":"paid"},{"id":"o4","status":"cancelled"}]} | HTTP 200

ลืมใส่ ?status= แล้ว c.Query คืน "" เงียบ ๆ ไม่ error ไม่ 400 — ได้ order ทั้งหมดรวม cancelled กลับมา แทนที่จะได้ list ว่าง · ใช้ c.GetQuery (หัวข้อ 9) ถ้าต้องแยก “ไม่ส่งมา” ออกจาก “ส่งมาว่าง”

c.Param คืนค่าที่ URL-decode แล้ว — ส่ง /users/abc%20xyz เข้าไปได้ "abc xyz" (มีช่องว่างจริง)

ตอนลองเองแล้วเจอ 404 งง ๆ — ยิง POST /users/123 (route มีแค่ GET กับ DELETE) ได้ 404 ไม่ใช่ 405 เพราะ gin.New() ตั้ง HandleMethodNotAllowed = false เป็นค่าเริ่มต้น · เปิดเป็น true แล้วจะได้ 405 พร้อม header Allow: GET — ดูหัวข้อ 8 และ 9
