3 · Mockery — เขียนเทสด้วย EXPECT()

func TestDisplayName_OK(t *testing.T) {
    repo := NewMockUserRepository(t)      // ← t.Cleanup + AssertExpectations ให้เอง
    repo.EXPECT().GetByID(7).Return(&User{ID: 7, Name: "Somchai"}, nil)

    svc := NewService(repo)
    got, err := svc.DisplayName(7)

    assert.NoError(t, err)
    assert.Equal(t, "Mr. Somchai", got)
}

ไม่ต้องเขียน defer repo.AssertExpectations(t) แล้ว — constructor ลงทะเบียน t.Cleanup(func(){ mock.AssertExpectations(t) }) ไว้ให้ (นี่คือความต่างจาก v2)

อยากได้เขียนแบบนี้
ค่า argument ตรงเป๊ะrepo.EXPECT().GetByID(7).Return(u, nil)
ไม่สนใจ argumentrepo.EXPECT().Save(mock.Anything).Return(nil)
ตรวจ argument แบบมีเงื่อนไขmock.MatchedBy(func(u *User) bool { ... })
คืน error.Return(nil, ErrNotFound)
คำนวณค่า return ตอนถูกเรียก.RunAndReturn(func(id int) (*User, error) { ... })
จำกัดจำนวนครั้ง.Once() · .Times(n) — ไม่ใส่ = กี่ครั้งก็ได้
ยืนยันว่าไม่ถูกเรียกrepo.AssertNotCalled(t, "Save", mock.Anything)
สไตล์เดิมจาก v2repo.On("GetByID", 1).Return(u, nil) — ยังใช้ได้ ปนกันได้

EXPECT() เป็นแค่ syntactic sugar ที่ห่อ mock.On("ชื่อ method", args...) ไว้ · .Return() type-safe แต่ argument ยังรับ any เพื่อให้ส่ง mock.Anything ได้

5 · Testcontainers — โครงที่ก็อปไปใช้ได้

func StartPostgres(ctx context.Context) (*tcpostgres.PostgresContainer, time.Duration, error) {
    start := time.Now()
    ctr, err := tcpostgres.Run(ctx,
        "postgres:16-alpine",
        tcpostgres.WithDatabase("testdb"),
        tcpostgres.WithUsername("testuser"),
        tcpostgres.WithPassword("testpass"),
        tcpostgres.BasicWaitStrategies(),      // ← ห้ามลืม
    )
    return ctr, time.Since(start), err
}

func OpenGorm(ctx context.Context, ctr *tcpostgres.PostgresContainer) (*gorm.DB, string, error) {
    dsn, err := ctr.ConnectionString(ctx, "sslmode=disable")   // ← port สุ่มอยู่ในนี้
    if err != nil { return nil, "", err }
    db, err := gorm.Open(postgres.Open(dsn), &gorm.Config{
        Logger: logger.Default.LogMode(logger.Silent),
    })
    if err != nil { return nil, dsn, err }
    if err := db.AutoMigrate(&model.User{}); err != nil { return nil, dsn, err }
    return db, dsn, nil
}

รูปแบบ cleanup ที่ถูกต้อง

❌ leak ได้ 2 ทาง

ctr, err := tcpostgres.Run(ctx, ...)
if err != nil {
    t.Fatal(err)     // ออกก่อนถึง defer
}
defer ctr.Terminate(ctx)  // panic ถ้า ctr = nil

✅ nil-safe + defer มาก่อน

ctr, err := tcpostgres.Run(ctx, ...)
defer func() {
    _ = testcontainers.TerminateContainer(ctr)
}()
if err != nil {
    t.Fatal(err)
}

Run คืนค่าไม่เหมือนกันสองกรณี: image tag ผิด → ctr = nil · wait strategy timeout → ctr ≠ nil พร้อม error (container start ไปแล้วจริง ๆ)
testcontainers.TerminateContainer(ctr) จัดการได้ทั้งคู่ — เรียกกับ typed nil ก็คืน <nil> ไม่ panic

BasicWaitStrategies() คืออะไร

wait.ForLog("database system is ready to accept connections").WithOccurrence(2)
wait.ForListeningPort("5432/tcp")

ทำไมต้อง 2 ครั้ง — postgres restart ตัวเองหลัง initdb · ทำไมต้องรอ port — macOS/Windows มี docker port-proxy แยก ถ้าไม่รอเทสจะ flaky

6 · Testcontainers — sync.Once + testify suite

var (
    sharedOnce sync.Once
    sharedCtr  *tcpostgres.PostgresContainer
    sharedDB   *gorm.DB
    sharedErr  error
    SharedBoot time.Duration
)

func SharedDB(t *testing.T) (*gorm.DB, string) {
    t.Helper()
    sharedOnce.Do(func() {
        ctx := context.Background()
        var d time.Duration
        sharedCtr, d, sharedErr = StartPostgres(ctx)
        SharedBoot = d
        if sharedErr != nil { return }
        sharedDB, sharedDSN, sharedErr = OpenGorm(ctx, sharedCtr)
    })
    if sharedErr != nil { t.Fatalf("shared container failed: %v", sharedErr) }
    return sharedDB, sharedDSN
}

// TestMain ทำแค่นี้ — ห้าม boot container ก่อน m.Run()
func TestMain(m *testing.M) {
    code := m.Run()
    testutil.TerminateShared()
    os.Exit(code)
}

8 · Goose — โครงไฟล์ migration

template ที่ goose create ... sql สร้างให้มี แค่ 4 บรรทัด (ไม่มี StatementBegin/End มาให้ — ต้องพิมพ์เอง):

-- +goose Up
SELECT 'up SQL query';

-- +goose Down
SELECT 'down SQL query';

DDL ปกติ

-- +goose Up
CREATE TABLE products (
    id         BIGSERIAL PRIMARY KEY,
    name       TEXT NOT NULL,
    price      NUMERIC(10,2) NOT NULL DEFAULT 0,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- +goose Down
DROP TABLE products;

function / trigger — ต้องครอบ StatementBegin/End

-- +goose Up
ALTER TABLE products ADD COLUMN updated_at TIMESTAMPTZ NOT NULL DEFAULT now();

-- +goose StatementBegin
CREATE OR REPLACE FUNCTION set_updated_at() RETURNS trigger AS $$
BEGIN
    NEW.updated_at = now();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
-- +goose StatementEnd

CREATE TRIGGER trg_products_updated_at
BEFORE UPDATE ON products
FOR EACH ROW EXECUTE FUNCTION set_updated_at();

-- +goose Down
DROP TRIGGER trg_products_updated_at ON products;
DROP FUNCTION set_updated_at();
ALTER TABLE products DROP COLUMN updated_at;

goose แบ่ง statement ด้วย ; แบบง่าย ๆ ไม่รู้จัก dollar-quoted string — ครอบเฉพาะ block ที่มี ; ข้างในก็พอ statement อื่นในไฟล์เดียวกันปล่อยไว้ได้

CREATE INDEX CONCURRENTLY — ต้องปิด transaction

-- +goose NO TRANSACTION
-- +goose Up
CREATE INDEX CONCURRENTLY idx_items_sku ON items (sku);

-- +goose Down
DROP INDEX CONCURRENTLY idx_items_sku;

goose ห่อทุกไฟล์ด้วย transaction โดย default (ไฟล์ที่พังกลางทางจึง rollback ทั้งไฟล์ ไม่มี state ค้าง) แต่ Postgres ห้าม CONCURRENTLY อยู่ใน transaction block
ใส่ NO TRANSACTION แล้ว = ไฟล์นั้นไม่มี rollback อัตโนมัติ → ใส่ statement เดียวต่อไฟล์

seed master data = migration ตัวหนึ่ง

-- +goose Up
INSERT INTO categories (code, name) VALUES
    ('ELEC', 'Electronics'),
    ('BOOK', 'Books'),
    ('FOOD', 'Food & Drink');

-- +goose Down
DELETE FROM categories WHERE code IN ('ELEC','BOOK','FOOD');

รัน goose up ซ้ำ 3 รอบ ข้อมูลยังเป็น 3 แถว — ตัวที่กันซ้ำคือ goose_db_version ไม่ใช่ SQL (INSERT ธรรมดาไม่มี ON CONFLICT ก็ไม่ซ้ำ เพราะ goose ไม่รันไฟล์นั้นอีกเลย)

goose vs GORM AutoMigrate

งานAutoMigrategoose
สร้างตาราง / เพิ่มคอลัมน์✅ ได้✅ ได้
เปลี่ยน type คอลัมน์เดิม✅ ได้ (สั่ง ALTER ... USING ให้เอง)✅ เขียนเอง คุมได้
rename คอลัมน์❌ สร้างคอลัมน์ใหม่ ข้อมูลไม่ตามมา✅ RENAME COLUMN ข้อมูลอยู่ครบ

func getUser(ctx context.Context, rdb *redis.Client, id int) (User, string, error) {
    key := fmt.Sprintf("user:%d", id)

    val, err := rdb.Get(ctx, key).Result()
    if err == nil {                          // HIT
        var u User
        if err := json.Unmarshal([]byte(val), &u); err != nil {
            return User{}, "", err
        }
        return u, "HIT", nil
    }
    if !errors.Is(err, redis.Nil) {           // error จริง — ต้องแยกจาก cache miss
        return User{}, "", fmt.Errorf("redis error: %w", err)
    }

    u := queryDB(id)                          // MISS
    b, err := json.Marshal(u)                  // ← อย่าใส่ _ ทิ้ง error
    if err != nil { return User{}, "", err }
    if err := rdb.Set(ctx, key, b, 60*time.Second).Err(); err != nil {
        return User{}, "", err
    }
    return u, "MISS", nil
}

cache stampede

var mu sync.Mutex

func getGuarded(ctx context.Context, rdb *redis.Client) string {
    if v, err := rdb.Get(ctx, "hot").Result(); err == nil { return v }
    mu.Lock()
    defer mu.Unlock()
    // เช็คซ้ำหลังได้ lock — คนอื่นอาจเติม cache ไปแล้ว
    if v, err := rdb.Get(ctx, "hot").Result(); err == nil { return v }
    v := queryDB()
    rdb.Set(ctx, "hot", v, 60*time.Second)
    return v
}
