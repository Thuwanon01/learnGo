บทเรียน 12 · Day 4 · Error Handling

Error Architecture: error คือข้อมูล ไม่ใช่ข้อความ

~16 นาที · ต้องมี: บทเรียน 05, บทเรียน 10

บทเรียน 05 สอนว่า Go คืน error เป็นค่า บทนี้ไปต่ออีกขั้น: error ไม่ได้มีไว้แค่ print — มันคือข้อมูลที่ควบคุมการทำงานของระบบ ตอบ 404 หรือ 500 · retry หรือยอมแพ้ · แสดงข้อความอะไรให้ผู้ใช้ ทั้งหมดนี้ตัดสินจาก error ที่ลอยขึ้นมา

1 · error เป็นแค่ interface ที่มี method เดียว

// นิยามจริงใน Go — built-in ไม่ต้อง import อะไร
type error interface {
    Error() string
}

แค่นั้นจริง ๆ อะไรก็ตามที่มี method Error() string คือ error ได้ทันที — ตรงตาม duck typing จาก บทเรียน 10 แปลว่าคุณสร้าง error ที่พก field อะไรก็ได้ ติดไปด้วย 

type ValidationError struct {
    Field   string
    Message string
}

func (e ValidationError) Error() string {
    return fmt.Sprintf("%s: %s", e.Field, e.Message)
}

func ValidateAge(age int) error {
    if age < 0 {
        return ValidationError{
            Field:   "age",
            Message: "must not be negative",
        }
    }
    return nil
}

ตอนนี้ผู้เรียกได้มากกว่าข้อความ — ได้ Field ไปใช้ highlight ช่องกรอกที่ผิดในหน้าเว็บได้

2 · errors.New() ต่างจาก fmt.Errorf() ยังไง

errors.New()fmt.Errorf()
ใช้เมื่อข้อความคงที่ ไม่มีตัวแปรต้องแทรกข้อมูล หรือห่อ error เดิม
ตัวอย่างerrors.New("insufficient balance")fmt.Errorf("read %q: %w", name, err)
ห่อ error ได้ไหม❌✅ ด้วย %w
เทียบด้วย ==✅ (จึงทำ sentinel ได้)❌ สร้างใหม่ทุกครั้ง
ระวังตอนพิมพ์ตามสไลด์

สไลด์หน้านี้แสดง func Withdraw(balance, amount int) error แล้วขึ้นบรรทัดใหม่ค่อยเปิด { — ถ้าพิมพ์ตามนั้นตรง ๆ จะ compile ไม่ผ่าน ผมรันแล้วได้ syntax error: unexpected semicolon or newline before { เพราะ Go เติม semicolon ท้ายบรรทัดให้อัตโนมัติ ปีกกาเปิดต้องอยู่บรรทัดเดียวกับ func เสมอ (น่าจะเป็นแค่การตัดบรรทัดตอนทำสไลด์ แต่ถ้าลอกไปจะงงว่าพังเพราะอะไร)

errors.New() — ข้อความคงที่

func Withdraw(balance, amount int) error {
    if amount > balance {
        return errors.New(
            "insufficient balance",
        )
    }
    return nil
}

fmt.Errorf() — เติมข้อมูล/ห่อ

func LoadConfig(filename string) error {
    data, err := os.ReadFile(filename)
    if err != nil {
        return fmt.Errorf(
            "read config %q: %w", filename, err,
        )
    }
    _ = data
    return nil
}

3 · Sentinel Error — error ที่มีชื่อให้เทียบได้

package user

// นิยมประกาศระดับ package และตั้งชื่อขึ้นต้นด้วย Err
var (
    ErrUserNotFound     = errors.New("user not found")
    ErrEmailAlreadyUsed = errors.New("email already used")
    ErrPermissionDenied = errors.New("permission denied")
)

ประโยชน์คือผู้เรียกตัดสินใจจาก error ได้ โดยไม่ต้องเทียบข้อความ: 

if errors.Is(err, user.ErrUserNotFound) {
    // ตอบ 404
} else if errors.Is(err, user.ErrPermissionDenied) {
    // ตอบ 403
}
ห้ามเด็ดขาด 
if err.Error() == "user not found" {   // ✗ อย่าทำ

ข้อความ error คือสิ่งที่มนุษย์อ่าน ไม่ใช่ API วันไหนมีคนแก้เป็น "User not found" (ตัว U ใหญ่) หรือใส่ id ต่อท้าย โค้ดคุณจะพังเงียบ ๆ โดยไม่มี compile error เตือน

4 · Error Chain — ห่อ error โดยไม่ทำต้นตอหาย

ระบบจริงมีหลายชั้น แต่ละชั้นควรเติมบริบทของตัวเองลงไป โดยไม่กลบข้อมูลของชั้นล่าง 

// Repository:
func (r UserRepository) Find(id string) (User, error) {
    user, err := queryDatabase(id)
    if err != nil {
        return User{}, fmt.Errorf("query user: %w", err)
    }
    return user, nil
}

// Service: เติมบริบทอีกชั้น
func (s UserService) GetProfile(id string) (User, error) {
    user, err := s.repository.Find(id)
    if err != nil {
        return User{}, fmt.Errorf("get profile %s: %w", id, err)
    }
    return user, nil
}
ข้อความสุดท้ายที่ handler เห็น:

  get profile U-101: query user: user not found
  └──── Service ────┘└─ Repository ─┘└─ ต้นตอ ─┘

แต่ละชั้นเติมคำนำหน้า ต้นตอยังอยู่ท้ายสุดเสมอ
และ errors.Is() ยังมองทะลุถึงต้นตอได้
err := fmt.Errorf("query user: %v", ErrUserNotFound)
                              //     ↑ %v ไม่ใช่ %w

fmt.Println(errors.Is(err, ErrUserNotFound))

พิมพ์ true หรือ false?

false — ผมรันทดสอบแล้ว

%v แค่แปลง error เป็นข้อความ แล้วโยนตัว error เดิมทิ้ง — ข้อความสุดท้ายอ่านเหมือนกันเป๊ะ แต่โซ่ขาด ส่วน %w เก็บ error เดิมไว้ข้างในให้ errors.Is ตามไปเจอ
นี่คือบั๊กที่หาไม่เจอที่สุดในเรื่อง error เพราะ log สวยงามทุกอย่าง แต่ errors.Is คืน false เงียบ ๆ

5 · errors.Is กับ errors.As — คนละงานกัน

errors.Is(err, target)errors.As(err, &target)
ถามว่า"ในโซ่นี้มี error ตัวนี้ ไหม""ในโซ่นี้มี error ชนิดนี้ ไหม"
คู่กับsentinel (errors.New)custom error struct
ได้อะไรกลับbool อย่างเดียวbool + ตัว error จริงพร้อม field
อาร์กิวเมนต์ 2ตัว errorpointer ไปยังตัวแปร (ต้องมี &)
// errors.Is — เช็คว่าใช่ตัวนั้นไหม
err := FindUser("U-101")
if errors.Is(err, ErrUserNotFound) {
    fmt.Println("show 404 page")
}
// errors.As — ดึงตัวจริงออกมาใช้ field
var validationErr *ValidationError

err := ValidateInput()
if errors.As(err, &validationErr) {
    fmt.Println("invalid field:", validationErr.Field)
    //                              ↑ ได้ field มาใช้ต่อ
}

จำง่าย ๆ: Is = "ใช่อันนี้ไหม" ตอบได้แค่ใช่/ไม่ใช่ · As = "เป็นชนิดนี้ไหม ถ้าใช่ขอตัวจริงด้วย"
สังเกตว่า errors.As ก็คือ type assertion จาก บทเรียน 11 ที่เดินตามโซ่ให้ด้วยนั่นเอง

6 · Custom Error Struct — แบบที่ใช้จริงในโปรดักชัน

type AppError struct {
    Op         string   // เกิดที่ไหน — "UserService.GetUser"
    Code       string   // รหัสให้เครื่องอ่าน — "USER_NOT_FOUND"
    HTTPStatus int      // ตอบ client ยังไง — 404
    Message    string   // ข้อความให้ "ผู้ใช้" อ่าน
    Err        error    // error เดิมที่ห่อไว้
}

func (e *AppError) Error() string {
    if e.Err != nil {
        return fmt.Sprintf("%s: %s: %v", e.Op, e.Message, e.Err)
    }
    return fmt.Sprintf("%s: %s", e.Op, e.Message)
}

// ⚠️ ขาดอันนี้ไม่ได้ — ไม่งั้น errors.Is/As มองทะลุไม่ได้
func (e *AppError) Unwrap() error {
    return e.Err
}

ใช้ที่ชั้น service

func GetUser(id string) (User, error) {
    user, err := repository.FindByID(id)
    if err != nil {
        if errors.Is(err, ErrUserNotFound) {
            return User{}, &AppError{
                Op:         "UserService.GetUser",
                Code:       "USER_NOT_FOUND",
                HTTPStatus: http.StatusNotFound,
                Message:    "ไม่พบผู้ใช้งาน",
                Err:        err,
            }
        }
        return User{}, &AppError{
            Op:         "UserService.GetUser",
            Code:       "INTERNAL_ERROR",
            HTTPStatus: http.StatusInternalServerError,
            Message:    "ไม่สามารถโหลดข้อมูลผู้ใช้ได้",
            Err:        err,
        }
    }
    return user, nil
}

ใช้ที่ชั้น handler

user, err := service.GetUser(id)
if err != nil {
    var appErr *AppError

    if errors.As(err, &appErr) {
        http.Error(w, appErr.Message, appErr.HTTPStatus)
        return
    }

    // error ที่ไม่รู้จัก — อย่าเผยรายละเอียดออกไป
    http.Error(w, "internal server error", http.StatusInternalServerError)
    return
}

ผลรันจริงของโครงนี้:

err.Error()  = UserService.GetUser: ไม่พบผู้ใช้งาน: query user U-101: user not found
errors.Is(err, ErrUserNotFound)  = true      ← ทะลุถึงต้นตอ
errors.As(err, &appErr)          = true
appErr.Code = USER_NOT_FOUND | HTTPStatus = 404
สิ่งที่โครงนี้แก้ให้

แยกข้อความสำหรับผู้ใช้ ออกจากข้อมูลสำหรับ developer — ผู้ใช้เห็น "ไม่พบผู้ใช้งาน" ส่วน log เก็บโซ่เต็ม ๆ ว่าพังตรงไหน และ handler ตัดสิน HTTP status ได้โดยไม่ต้องรู้จัก database เลยสักนิด

7 · กับดักร้ายแรง: nil ที่ไม่ nil

func doWork() error {
    var e *AppError = nil
    return e            // คืน pointer ที่เป็น nil
}

if doWork() == nil {
    fmt.Println("ไม่มี error")
} else {
    fmt.Println("มี error!")
}

พิมพ์อะไรออกมา?

มี error! — ผมรันทดสอบแล้ว doWork() == nil ได้ false

ย้อนกลับไปดูรูปกล่องใน บทเรียน 11 — interface เก็บ สองช่อง:

┌──────────────┬────────────┐
│   *AppError  │    nil     │   ← ช่อง type ไม่ว่าง!
└──────────────┴────────────┘
interface จะ == nil ก็ต่อเมื่อ ทั้งสองช่อง ว่าง

ทางแก้: อย่าประกาศตัวแปรเป็น *AppError แล้วคืนมันไปตรง ๆ ให้ return nil ตรง ๆ ในเส้นทางที่สำเร็จ: 

func doWork() error {
    if somethingWrong {
        return &AppError{...}
    }
    return nil          // ✓ nil ของแท้
}

8 · ทดสอบความจำ

1. ใช้ %v แทน %w ใน fmt.Errorf มีผลยังไง?

นี่คือกับดักที่อันตรายที่สุด เพราะ log ออกมาสวยงามเหมือนเดิมทุกอย่าง แต่ errors.Is คืน false เงียบ ๆ → handler ตอบ 500 แทนที่จะเป็น 404 %w เท่านั้นที่เก็บ error เดิมไว้ในโซ่

2. custom error struct ขาด method Unwrap() จะเป็นยังไง?

errors.Is/As เดินตามโซ่โดยเรียก Unwrap() ไปเรื่อย ๆ — ไม่มี Unwrap() = โซ่จบตรงนั้น field Err ยังอยู่ครบ แต่ไม่มีใครตามไปดูได้

3. อยากได้ Field จาก *ValidationError ในโซ่ ใช้อะไร?

errors.Is ตอบได้แค่ bool เอา field มาไม่ได้ · errors.As(err, &validationErr) เดินตามโซ่ให้แล้ว ยัดตัวจริงลงในตัวแปรที่คุณส่ง pointer ไป → ใช้ validationErr.Field ได้เลย · อย่าลืม &

4. คืน var e *AppError = nil จากฟังก์ชันที่ประกาศคืน error?

interface เก็บ type + value — ช่อง type มี *AppError อยู่ ไม่ว่าง จึงไม่เท่ากับ nil ทั้งที่ค่าข้างในเป็น nil วิธีกัน: เขียน return nil ตรง ๆ ในเส้นทางที่สำเร็จ อย่าคืนตัวแปรที่เป็น pointer type

9 · ลงมือทำ

ภารกิจ: ระบบค้นหาสินค้าและจัดการ Error Chain (workshop จากสไลด์)

  1. ประกาศ sentinel: var ErrProductNotFound = errors.New("product not found")
  2. ชั้น repository: findProduct(code string) (Product, error) — ถ้าไม่พบให้คืน ErrProductNotFound
  3. ชั้น service: getProduct(code string) (Product, error) — ห่อด้วย fmt.Errorf("find product %q: %w", code, err) แล้วครอบด้วย &AppError{Code: "PRODUCT_NOT_FOUND", Message: "ไม่พบสินค้าที่ต้องการ", Err: ...}
    ภารกิจนี้ใช้ AppError แบบย่อ (แค่ 3 field) และ Error() คืน fmt.Sprintf("%s: %v", e.Message, e.Err) — เพิ่ม Op/HTTPStatus ทีหลังได้
  4. อย่าลืมเขียน func (e *AppError) Unwrap() error
  5. ชั้น main: ตรวจด้วย errors.Is() และ errors.As() ให้ได้ผลลัพธ์: 
    พบ ErrProductNotFound ใน Error Chain
code: PRODUCT_NOT_FOUND
message: ไม่พบสินค้าที่ต้องการ
error: ไม่พบสินค้าที่ต้องการ: find product "P999": product not found
  6. ห้ามตรวจ error ด้วยการเทียบข้อความ string — กฎจากสไลด์
  7. ท้าทาย 1: เปลี่ยน %w เป็น %v ในข้อ 3 แล้วรันใหม่ — ข้อความเปลี่ยนไหม? errors.Is เปลี่ยนไหม? นี่คือการพิสูจน์กับดักในหัวข้อ 4 ด้วยมือตัวเอง
  8. ท้าทาย 2: ลบ Unwrap() ออก แล้วดูว่า errors.Is กับ errors.As อันไหนพัง
    สปอยล์ผลรันจริง: errors.Is → false แต่ errors.As → true เหมือนเดิม ทำไม As ถึงยังรอด? คำใบ้: err ตัวนอกสุดคือ *AppError อยู่แล้ว ไม่ต้องเดินโซ่ก็เจอ
