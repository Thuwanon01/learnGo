บทเรียน 08 · Resource Cleanup

defer: เปิดแล้วปิดให้ครบทุกทางออก

~12 นาที · ต้องมี: บทเรียน 02

ฟังก์ชันหนึ่งมี return ได้ 5 จุด ถ้าต้องปิด file ทุกจุด คุณต้อง เขียน Close() 5 ครั้ง — และวันหนึ่งจะมีคนเพิ่ม return ที่ 6 แล้วลืม defer ทำให้ปัญหานี้หายไปเลย ด้วยกฎแค่ 3 ข้อ ซึ่งมีข้อหนึ่งที่หลอกคนเกือบทุกคน

1 · ปัญหาที่ defer แก้

ลองดูโค้ดแบบไม่มี defer ก่อน — นับดูว่า Close() ต้องเขียนกี่ที

func readFile(name string) error {
    f, err := os.Open(name)
    if err != nil { return err }

    data := make([]byte, 100)
    n, err := f.Read(data)
    if err != nil {
        f.Close()          // ← 1
        return err
    }
    if n == 0 {
        f.Close()          // ← 2  (ลืมข้อนี้ = file leak)
        return errEmpty
    }
    f.Close()              // ← 3
    return nil
}

เขียนใหม่ด้วย defer:

func readFile(name string) error {
    f, err := os.Open(name)
    if err != nil {
        return err        // ยังไม่ได้เปิด → ไม่ต้องปิด
    }
    defer f.Close()       // ✓ เขียนครั้งเดียว ติดกับ Open เลย

    data := make([]byte, 100)
    n, err := f.Read(data)
    if err != nil { return err }   // Close ทำงานที่นี่
    if n == 0      { return errEmpty } // และที่นี่
    return nil                        // และที่นี่
}
คุณค่าที่แท้จริง

ไม่ใช่แค่ "โค้ดสั้นลง" แต่คือ Open กับ Close อยู่ติดกันในสายตา — reviewer เห็นสองบรรทัดนี้คู่กันแล้วรู้ทันทีว่า ไม่รั่ว ไม่ต้องไล่อ่านทั้งฟังก์ชัน

2 · กฎ 3 ข้อ (จาก blog ทางการ)

Defer, Panic, and Recover ระบุกฎไว้ 3 ข้อพอดี ท่องให้ได้ทั้งสามข้อ

กฎ 1 — argument ถูกประเมิน ตอนบรรทัด defer ไม่ใช่ตอนทำงาน

"A deferred function's arguments are evaluated when the defer statement is evaluated." 
func a() {
    i := 0
    defer fmt.Println(i)
    i++
    return
}

พิมพ์เลขอะไรออกมา?

0 — ไม่ใช่ 1

ค่า i ถูก "แช่แข็ง" เป็น 0 ตั้งแต่วินาทีที่เจอบรรทัด defer การ i++ ทีหลังไม่มีผลอะไรเลย นี่คือกฎที่หลอกคนมากที่สุดในสามข้อ

ถ้าอยากได้ค่า ล่าสุด ให้ห่อด้วย closure แทน:

i := 0
defer fmt.Println(i)                          // พิมพ์ 0  — ส่งค่าเข้าไป
defer func() { fmt.Println(i) }()          // พิมพ์ 1  — closure อ่านตอนทำงาน
i++

วิธีจำ: defer f(x) → x ถ่ายรูปเก็บไว้เลย · defer func(){ …x… }() → ไม่มี argument จึงไม่มีอะไรให้ถ่ายรูป ต้องไปอ่านตัวแปรจริงตอนทำงาน

กฎ 2 — ทำงานแบบ LIFO หลังฟังก์ชัน return

"Deferred function calls are executed in Last In First Out order after the surrounding function returns." 
func b() {
    for i := 0; i < 4; i++ {
        defer fmt.Print(i)
    }
}

พิมพ์อะไรออกมา?

3210

กองซ้อนกันเหมือน stack — ตัวที่ประกาศทีหลังทำงานก่อน (และสังเกตว่ากฎ 1 ทำงานด้วย: แต่ละรอบถ่ายรูปค่า i ของรอบนั้นไว้)

LIFO ไม่ใช่เรื่องบังเอิญ — dependency ต้องคลายย้อนกลับเสมอ ดูตัวอย่างจริงจากงาน backend: 

func processOrder(db *sql.DB) error {
    logFile, _ := os.OpenFile("log.txt", os.O_APPEND, 0644)
    defer logFile.Close()   // ประกาศก่อน → ทำทีหลังสุด

    tx, _ := db.Begin()
    defer tx.Rollback()    // ประกาศทีหลัง → ทำก่อน ✓ ถูกลำดับ

    if _, err := tx.Exec("INSERT INTO orders ..."); err != nil {
        return err          // rollback อัตโนมัติ
    }
    return tx.Commit()     // commit แล้ว Rollback ทีหลังไม่มีผล
}

defer tx.Rollback() ทันทีหลัง Begin() แปลว่า "ถ้าไม่ commit สำเร็จ ให้ย้อนกลับ" — ไม่ต้องเขียน rollback ซ้ำในทุก error branch

กฎ 3 — deferred function แก้ named return value ได้

"Deferred functions may read and assign to the returning function's named return values." 
func c() (i int) {        // ← ตั้งชื่อ return value ว่า i
    defer func() { i++ }()
    return 1
}

ฟังก์ชันนี้คืนค่าอะไร?

2

return 1 ทำงาน 2 ขั้น: (1) เอา 1 ใส่ i → (2) รัน deferred → i++ เป็น 2 → (3) ค่อยคืนค่าออกไปจริง ต้องตั้งชื่อ return value เท่านั้นถึงจะทำได้

ใช้ทำอะไรจริง — ห่อ error ให้มีบริบทโดยไม่ต้องแก้ทุกจุด return:

func loadConfig(path string) (err error) {
    defer func() {
        if err != nil {
            err = fmt.Errorf("loadConfig %s: %w", path, err)
        }
    }()

    // ...มี return err อยู่ 5 จุด ทุกจุดจะถูกห่อให้อัตโนมัติ
}

3 · 2 กับดักที่ต้องรู้

กับดัก 1 — scope คือ "ฟังก์ชัน" ไม่ใช่ block

defer ในลูปจะ กองสะสมจนจบฟังก์ชัน ไม่ใช่จบทุกรอบ

// ❌ เปิด file 1000 ตัวค้างพร้อมกัน
for _, name := range files {
    f, _ := os.Open(name)
    defer f.Close()
}

// ✓ ครอบด้วยฟังก์ชันย่อย → จบทุกรอบ
for _, name := range files {
    func() {
        f, _ := os.Open(name)
        defer f.Close()
        // ...
    }()
}

Go 1.22+ มี go vet ที่ช่วยเตือนบางกรณี แต่อย่าพึ่งมัน

กับดัก 2 — os.Exit และ log.Fatal ไม่รัน defer 
defer fmt.Println("จะไม่ถูกพิมพ์")
log.Fatal("boom")   // = log.Print + os.Exit(1) → ตายทันที

ต่างจาก panic ซึ่ง รัน defer ครบทุกตัว ก่อนล้ม — นั่นแหละที่ทำให้ defer เป็นเครื่องมือกู้คืน resource ที่ปลอดภัยที่สุด

4 · ทดสอบความจำ

1. x := 5; defer fmt.Println(x); x = 99 — พิมพ์อะไร?

กฎ 1 — argument ถ่ายรูปเก็บตอนบรรทัด defer ถ้าอยากได้ 99 ต้องเขียน defer func(){ fmt.Println(x) }()

2. เขียน defer A() แล้ว defer B() ตามลำดับ อะไรทำงานก่อน?

กฎ 2 — LIFO เหมือน stack ออกแบบมาเพื่อให้คลาย dependency ย้อนกลับได้ถูกต้อง (เปิด db → เปิด tx → ปิด tx → ปิด db)

3. โปรแกรมเจอ panic กลางทาง — defer ที่ประกาศไว้แล้วจะ…

panic คลาย stack ทีละชั้น และรัน deferred ของแต่ละชั้นครบ นี่คือเหตุผลที่ defer mu.Unlock() ปลอดภัยกว่าการเรียก mu.Unlock() เอง — ต่อให้โค้ดตรงกลาง panic ก็ยังปลดล็อกให้

ข้อยกเว้นเดียวคือ os.Exit / log.Fatal ที่ตายทันทีไม่คลาย stack

4. defer จะแก้ค่าที่ฟังก์ชันคืนออกไปได้ ก็ต่อเมื่อ…

กฎ 3 — ต้องเป็น named return value เช่น func f() (err error) ถึงจะมีตัวแปรให้ deferred function ไปแก้ได้ ถ้าเขียน func f() error เฉย ๆ จะไม่มีชื่อให้อ้างถึง

5 · ลงมือทำ

ภารกิจ: ทำ utility จับเวลา

  1. รันโค้ดกฎ 1, 2, 3 ทั้งสามอันใน Playground ให้เห็นผลจริงด้วยตา
  2. แก้ตัวอย่างกฎ 1 ให้พิมพ์ 1 แทน 0 โดยห้ามย้ายบรรทัด defer
  3. สร้างฟังก์ชันนี้แล้วอธิบายให้ได้ว่าทำไมต้องมีวงเล็บ สองชุด: 
    func trackTime(name string) func() {
    start := time.Now()
    return func() {
        fmt.Printf("%s took %v\n", name, time.Since(start))
    }
}

func work() {
    defer trackTime("work")()   // ← สองชุด!
    time.Sleep(500 * time.Millisecond)
}
  4. ลองลบวงเล็บชุดหลังออก แล้วดูว่าเกิดอะไรขึ้น (คำใบ้: มันจะเงียบมาก)
  5. ท้าทาย: เขียนฟังก์ชันที่ใช้กฎ 3 ห่อ error — func doWork() (err error) ที่ไม่ว่าจะ return จากจุดไหน error จะถูกเติมคำว่า "doWork: " ข้างหน้าเสมอ
ชัยชนะของบทนี้

ต่อไปทุกครั้งที่พิมพ์ Open, Begin, Lock หรือ Dial มือคุณจะพิมพ์บรรทัด defer ต่อทันทีโดยไม่ต้องคิด — นั่นคือนิสัยที่แยกโค้ดที่รั่วออกจากโค้ดที่ไม่รั่ว
