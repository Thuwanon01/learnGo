2 · Dependencies — oauth2 กับ go-oidc ต่างกันยังไง

สไลด์หน้า 49 มีแค่สองบรรทัดนี้ ไม่มีคำอธิบายอะไรเลย:

go get golang.org/x/oauth2
go get github.com/coreos/go-oidc/v3

คนใหม่จะงงว่าทำไมต้องลงสองตัว ในเมื่อมันก็เรื่องเดียวกัน คำตอบคือมันทำงานคนละครึ่งของกระบวนการ และไม่มีตัวไหนทำแทนกันได้:

Libraryรับผิดชอบอะไรถ้าไม่มีตัวนี้จะต้องเขียนเองว่าอะไร
golang.org/x/oauth2 ขั้นตอนแลกของ — ประกอบ URL พาผู้ใช้ไปหน้า login · เอา code ที่ได้กลับมา ยิงไป token_endpoint เพื่อแลกเป็น token · เก็บ token ไว้ต่ออายุให้ ต้องประกอบ query string เอง ยิง POST เอง แกะ JSON เอง จัดการ refresh เอง
github.com/coreos/go-oidc/v3 ขั้นตอนตรวจของ — อ่าน discovery document ให้ · ไปโหลด public key จาก jwks_uri มาเอง · verify ลายเซ็นของ id_token และเช็ค iss/aud/วันหมดอายุให้ ต้องดึง JWKS เอง จับคู่ kid เอง ประกอบ RSA public key เอง verify เอง แล้วเช็ค claim เองทีละตัว
🏦 เทียบให้เห็นภาพ — คนเดินเรื่องที่เคาน์เตอร์ กับเครื่องส่องแบงก์ปลอม

นึกถึงตอนไปทำธุระที่เคาน์เตอร์ธนาคาร งานมันแยกเป็นสองคนละคนอยู่แล้ว:
· คนเดินเรื่อง — รู้ว่าต้องไปช่องไหน ยื่นใบคิวตรงไหน เอาใบคิวไปแลกเป็นของกลับมายังไง
· เครื่องส่องแบงก์ปลอม — ไม่สนใจว่าคุณเดินมาจากช่องไหน มันมีหน้าที่เดียวคือเอาของที่ได้มา ส่องดูว่าเป็นของจริงหรือของปลอม

oauth2 คือคนเดินเรื่อง — ประกอบ URL พาผู้ใช้ไปหน้า login แล้วเอา code ที่ได้กลับมาไปแลกเป็น token
go-oidc คือเครื่องส่อง — เอา id_token ที่แลกมาได้ไปตรวจลายเซ็น ว่า Keycloak เซ็นมาจริงไหม หมดอายุหรือยัง
คนเดินเรื่องไม่มีเครื่องส่อง เครื่องส่องก็เดินเรื่องแทนไม่ได้ — ต้องมีทั้งคู่ ถึงได้ลงสองบรรทัด

ที่ไม่เหมือนกัน: เครื่องส่องแบงก์ที่เคาน์เตอร์เสียบปลั๊กก็ทำงานได้เลย ตรวจจากตัวแบงก์อย่างเดียว แต่ go-oidc ต้องออกไปโหลด public key จาก jwks_uri ผ่านเน็ตก่อน ถึงจะตรวจได้ — เน็ตไปไม่ถึง Keycloak เมื่อไหร่ การ verify ก็ทำไม่ได้ทันที
เพราะแบบนี้ oidc.NewProvider ในหัวข้อ 3 ถึงยิง HTTP ออกไปตั้งแต่ตอนสร้าง ไม่ใช่ฟังก์ชันที่แค่ตั้งค่าไว้เฉย ๆ

📖 ศัพท์ที่ต้องรู้ก่อนไปต่อ (จากบทเรียน 50–51 แต่ทวนให้)
· id_token = JWT ที่บอกว่า "คนนี้คือใคร" — มีชื่อ อีเมล และ sub อยู่ข้างใน เอาไว้ให้แอปเรารู้จักผู้ใช้ ไม่ใช่ เอาไว้ยิง API
· access_token = JWT ที่บอกว่า "คนนี้ทำอะไรได้บ้าง" เอาไว้แนบไปกับ request
· jwks_uri = URL ที่ Keycloak แจก public key ของตัวเอง ใครก็โหลดได้ ไม่ใช่ความลับ — เอาไว้ตรวจว่า token ใบนี้ Keycloak เซ็นมาจริงหรือมีคนปลอม
· sub (subject) = รหัสประจำตัวผู้ใช้ที่ Keycloak ตั้งให้ เป็น UUID และไม่เปลี่ยนตลอดชีพของ user คนนั้น — จำคำนี้ไว้ เดี๋ยวหัวข้อ 7 ใช้

3 · โค้ด OIDC ครบ flow — ที่ compile ผ่านจริง

สไลด์บอกแค่ให้ลง library แต่ไม่ได้ให้โค้ดมา ผมเลยเขียนให้ครบทั้งเส้นแล้ว build ผ่านและ vet ผ่านจริง (ผลอยู่ในกล่องข้างบน)
ก่อนดูโค้ด ดูลำดับก่อน เพราะโค้ดจะอ่านง่ายขึ้นมาก — มันมีแค่ 6 ขั้น:

#ฟังก์ชันเกิดอะไรขึ้น
1oidc.NewProvider(ctx, issuer) ยิง HTTP ไปที่ issuer + /.well-known/openid-configuration ตอนนั้นเลย แล้วจำ URL ทุกตัวไว้
2provider.Endpoint() คืน AuthURL กับ TokenURL ที่เพิ่งอ่านมา — เราไม่ต้อง hardcode URL เอง
3oauth2.Config{…} ก้อนตั้งค่า: client id/secret, redirect URL, scope และ endpoint จากข้อ 2
4cfg.AuthCodeURL(state) ประกอบ URL หน้า login ให้ แล้วเรา redirect ผู้ใช้ไป
5cfg.Exchange(ctx, code) ยิง POST ไป token_endpoint เอา code แลกเป็น token — ขั้นนี้เป็น server-to-server
6verifier.Verify(ctx, rawIDToken) โหลด public key จาก jwks_uri มาตรวจลายเซ็นของ id_token แล้วคืน token ที่แกะได้

ทีนี้โค้ดจริง — ไฟล์ auth/auth.go ก้อนตั้งค่า
ก้อนนี้กับก้อนถัดไปคือไฟล์เดียวกัน วางต่อกันตามลำดับแล้ว go build ./... ผ่านเลย (ต้องอยู่ในโฟลเดอร์ auth/ ให้ตรงกับบรรทัด package) 

package auth

import (
	"context"
	"errors"

	"github.com/coreos/go-oidc/v3/oidc"
	"golang.org/x/oauth2"
)

// Claims คือ field ที่เราสนใจจาก id_token ของ Keycloak
type Claims struct {
	Sub               string `json:"sub"`
	Email             string `json:"email"`
	Name              string `json:"name"`
	PreferredUsername string `json:"preferred_username"`
}

type Authenticator struct {
	provider *oidc.Provider
	oauth2   oauth2.Config
	verifier *oidc.IDTokenVerifier
}

func New(ctx context.Context, issuer, clientID, clientSecret, redirectURL string) (*Authenticator, error) {
	// ① ยิงไปอ่าน discovery document ตรงนี้เลย — ถ้า Keycloak ยังไม่ขึ้น บรรทัดนี้ error ทันที
	provider, err := oidc.NewProvider(ctx, issuer)
	if err != nil {
		return nil, err
	}

	cfg := oauth2.Config{
		ClientID:     clientID,
		ClientSecret: clientSecret,
		RedirectURL:  redirectURL,
		Endpoint:     provider.Endpoint(),           // ② ไม่ hardcode URL
		Scopes:       []string{oidc.ScopeOpenID, "profile", "email"},
	}

	return &Authenticator{
		provider: provider,
		oauth2:   cfg,
		verifier: provider.Verifier(&oidc.Config{ClientID: clientID}),
	}, nil
}
⚠️ oidc.ScopeOpenID ไม่ใช่ของประดับ — ขาดแล้วไม่มี id_token

oidc.ScopeOpenID คือค่าคงที่ที่มีค่าเป็นสตริง "openid"
ถ้าไม่ใส่ scope ตัวนี้ Keycloak จะถือว่านี่เป็นคำขอแบบ OAuth2 ธรรมดา ไม่ใช่ OIDC — มันจะคืน access_token มาให้ตามปกติ แต่จะไม่มี id_token ในคำตอบ
อาการที่คุณจะเจอคือโค้ดขั้นที่ 6 พังด้วยข้อความว่าหา id_token ไม่เจอ ทั้งที่ทุกอย่างก่อนหน้านั้นดูสำเร็จหมด

ต่อด้วยสองฟังก์ชันที่ handler เรียกใช้จริง — ขาไปกับขากลับ (ตัว handler เองอยู่ท้ายหัวข้อ 9)
ก้อนนี้เขียนต่อท้ายไฟล์ auth/auth.go เดิม จึงไม่มีบรรทัด package และไม่มี import ซ้ำอีก — context กับ errors ที่ใช้ตรงนี้ ประกาศไว้ในก้อนบนแล้ว 

// LoginURL คือ URL ที่เรา redirect ผู้ใช้ไปหน้า login ของ Keycloak
func (a *Authenticator) LoginURL(state string) string {
	return a.oauth2.AuthCodeURL(state)
}

// Callback เอา code ที่ Keycloak ส่งกลับมาแลกเป็น token แล้ว verify id_token
func (a *Authenticator) Callback(ctx context.Context, code string) (*Claims, error) {
	tok, err := a.oauth2.Exchange(ctx, code)
	if err != nil {
		return nil, err
	}

	// ⚠️ id_token ไม่ได้อยู่ใน tok.AccessToken — ต้องดึงจาก Extra
	rawIDToken, ok := tok.Extra("id_token").(string)
	if !ok {
		return nil, errors.New("no id_token in token response")
	}

	idToken, err := a.verifier.Verify(ctx, rawIDToken)
	if err != nil {
		return nil, err
	}

	var c Claims
	if err := idToken.Claims(&c); err != nil {
		return nil, err
	}
	return &c, nil
}
⚠️ จุดที่คนพลาดมากที่สุดในโค้ดก้อนนี้: tok.Extra("id_token")

oauth2.Token ที่ Exchange คืนมามี field มาตรฐานอยู่ไม่กี่ตัว: AccessToken, TokenType, RefreshToken, Expiry
ไม่มี field ชื่อ IDToken เพราะ id_token เป็นของ OIDC ไม่ใช่ของ OAuth2 ซึ่ง library oauth2 ไม่รู้จัก — มันเก็บ field ที่ไม่รู้จักไว้ในถุงรวมแล้วให้หยิบด้วย tok.Extra(ชื่อ) ซึ่งคืน any จึงต้อง .(string) แปลงเอง
คนที่เพิ่งเริ่มมักเผลอเอา tok.AccessToken ไป Verify แล้วงงว่าทำไมชื่อกับอีเมลไม่ครบ — เพราะมันคนละใบกันจริง ๆ
และ Verify คืน สองค่า คือ (*oidc.IDToken, error) ยังไม่ใช่ข้อมูลผู้ใช้ — ต้องเรียก idToken.Claims(&dst) อีกทีถึงจะได้ field ออกมา

ทายก่อนรัน — ผมเขียนโปรแกรมเล็ก ๆ ที่เรียกแค่สามอย่าง โดยชี้ไป Google แทน Keycloak (เพราะ Google เป็น OIDC provider เหมือนกันและเปิดสาธารณะ ใครก็ยิงได้)
ก้อนข้างล่างเป็นเศษโค้ดที่ตัดมาเฉพาะใจความ ไม่ใช่ไฟล์เต็ม — ก๊อปไปวางเปล่า ๆ ยัง build ไม่ผ่าน ต้องมี package main · import (context, fmt, oidc, oauth2) · func main() และประกาศ ctx กับ err ครอบไว้ก่อน 

provider, _ := oidc.NewProvider(ctx, "https://accounts.google.com")
ep := provider.Endpoint()
fmt.Println("AuthURL  =", ep.AuthURL)
fmt.Println("TokenURL =", ep.TokenURL)

cfg := oauth2.Config{ClientID: "my-client-id", RedirectURL: "http://localhost:8080/auth/callback",
	Endpoint: ep, Scopes: []string{oidc.ScopeOpenID, "profile", "email"}}
fmt.Println("AuthCodeURL =", cfg.AuthCodeURL("state-abc123"))

verifier := provider.Verifier(&oidc.Config{ClientID: "my-client-id"})
_, err = verifier.Verify(ctx, "this-is-not-a-jwt")   // จงใจส่งขยะเข้าไป

ทายสองข้อ
1. TokenURL ของ Google จะขึ้นต้นด้วย https://accounts.google.com/... เหมือน AuthURL ไหม?
2. ตอนส่งสตริง "this-is-not-a-jwt" เข้า Verify — error ที่ได้จะเป็นข้อความของ go-oidc เอง หรือหลุดมาจาก library ชั้นล่าง?

ข้อ 1 — คนละโดเมนกัน AuthURL อยู่ที่ accounts.google.com แต่ TokenURL อยู่ที่ oauth2.googleapis.com
นี่คือเหตุผลที่ provider.Endpoint() มีค่ามาก: ถ้า hardcode เองคุณจะเดาผิดแน่นอน และเวลา provider ย้ายโดเมน โค้ดเราไม่ต้องแก้เลยเพราะมันอ่านใหม่ทุกครั้งที่ start

ข้อ 2 — error ไหลมาจากชั้นล่าง ข้อความมีสองท่อน: ท่อนหน้า oidc: malformed jwt: เป็นของ go-oidc ส่วนท่อนหลัง go-jose/go-jose: compact JWS format must have three parts มาจาก go-jose/v4 ตรง ๆ
มันกำลังบอกว่า JWT ที่ถูกต้องต้องมีสามท่อนคั่นด้วยจุด (header.payload.signature) แต่สตริงที่เราส่งไปไม่มีจุดเลย
นี่เป็นตัวอย่างที่ดีของเวลาอ่าน error ในโปรเจกต์จริง — ชื่อ library ที่อยู่หน้าข้อความคือเบาะแสว่าใครเป็นคนบ่น

สังเกต AuthCodeURL ด้วย: redirect_uri ถูก encode เป็น http%3A%2F%2Flocalhost%3A8080%2F… ให้อัตโนมัติ และ scope สามตัวถูกรวมด้วยเครื่องหมาย + — นี่คืองานที่ oauth2 ทำแทนเรา

🧭 state คืออะไร ทำไมต้องมี — state เป็นสตริงสุ่มที่เราสร้างตอนส่งผู้ใช้ไป login แล้ว Keycloak จะส่งกลับมาให้เหมือนเดิมตอน callback
หน้าที่ของมันคือเช็คว่า callback นี้มาจากคำขอที่เราเป็นคนเริ่มจริง ๆ ไม่ใช่ใครยิง URL callback ใส่ผู้ใช้ของเราเอง (การโจมตีแบบนี้เรียกว่า CSRF)
ในตัวอย่างข้างบนผมใส่ "state-abc123" ตายตัวเพราะเป็นแค่การทดสอบ — ของจริงต้องสุ่มใหม่ทุกครั้ง เก็บไว้ใน session แล้วเทียบตอนกลับมา
โค้ดที่ทำแบบนั้นจริงอยู่ท้ายหัวข้อ 9 (ฟังก์ชัน Login กับ Callback)

4 · ตั้ง goose ใน .env — และ DSN ในสไลด์ที่ต้องรู้ก่อนพิมพ์ตาม

ข ?database=wongnok ชื่อฐานข้อมูลตามมาตรฐาน PostgreSQL URI ต้องอยู่ในส่วน path คือ /wongnok ไม่ใช่ query parameter

พอแก้ตามที่ error ใบ้ (ย้าย ?database=wongnok ไปเป็น path /wongnok) แล้วสั่งใหม่ — ด่านสองถึงจะโผล่: could not translate host name "devp00l@localhost" ซึ่งไม่มีคำว่ารหัสผ่านอยู่ในข้อความเลย คุณจะไปนั่งไล่ดู DNS ไล่ดู /etc/hosts ไล่ดูว่า Postgres รันอยู่ไหม

กฎที่ควรจำติดตัว: อักขระพิเศษในรหัสผ่านต้อง URL-encode เสมอ — ตัวที่เจอบ่อยคือ @ → %40 · : → %3A · / → %2F · # → %23

✉️ เทียบให้เห็นภาพ — ที่อยู่บนหน้าซองที่เขียนติดกันเป็นบรรทัดเดียว

DSN คือที่อยู่ของฐานข้อมูลที่ยัดทุกอย่างไว้ในบรรทัดเดียว — ใครเข้า รหัสอะไร เครื่องไหน พอร์ตอะไร ฐานชื่อไหน แล้วใช้เครื่องหมายเป็นตัวคั่นว่าตรงไหนจบเรื่องอะไร · @ คือตัวคั่นที่แปลว่า "ก่อนหน้านี้คือชื่อกับรหัสผ่าน หลังจากนี้คือที่อยู่เครื่อง"

ทีนี้ลองนึกว่าเขียนหน้าซองว่า "สมชาย, ร้านข้าวมันไก่, ซอย 5" แล้วระบบเขาใช้ลูกน้ำเป็นตัวคั่นชื่อผู้รับกับที่อยู่พอดี
พนักงานคนหนึ่งอาจตัดที่ลูกน้ำตัวแรก (ผู้รับ = "สมชาย") อีกคนตัดที่ลูกน้ำตัวสุดท้าย (ผู้รับ = "สมชาย, ร้านข้าวมันไก่") — ข้อความเดียวกัน แต่ได้ผลคนละอย่าง
นี่คือสิ่งที่เกิดขึ้นจริงในหัวข้อนี้: รหัสผ่าน Pe@devp00l มี @ อยู่ในตัวเอง Go ตัดที่ @ ตัวสุดท้าย จึงรอด · libpq ที่ psql ใช้ตัดที่ตัวแรก จึงพัง

ที่ไม่เหมือนกัน: พนักงานไปรษณีย์อ่านแล้วเอะใจได้เองว่าอันไหนน่าจะเป็นชื่อคน แต่โปรแกรมไม่เดา มันทำตามกฎที่เขียนไว้ตายตัว และกฎของสองฝั่งนี้เขียนไว้คนละแบบจริง ๆ
ทางแก้จึงไม่ใช่ "เขียนให้อ่านง่ายขึ้น" หรือ "ครอบด้วยเครื่องหมายคำพูด" — แต่คือทำให้ไม่มี @ ดิบเหลืออยู่ในสตริงเลย ด้วยการเขียนเป็น %40 ซึ่งทั้งสองฝั่งอ่านตรงกันว่า "นี่คือตัวอักษร ไม่ใช่ตัวคั่น"

7 · ทำไมต้องมี uid ทั้งที่มี id อยู่แล้ว

ตารางนี้มีคอลัมน์ที่หน้าตาเหมือนรหัสประจำตัวอยู่สองตัว ซึ่งเป็นจุดที่คนอ่านสไลด์แล้วงงที่สุด
ทั้งสองตัวไม่ใช่ของซ้ำกัน มันเป็นรหัสของคนละระบบ:

 iduid
ใครเป็นคนตั้งค่า แอปเราเอง (UUID ที่เราสร้าง) Keycloak — คือค่า sub ที่อยู่ใน token
เอาไว้ทำอะไร เป็น primary key · ให้ตารางอื่น (recipes, ratings) ชี้มาด้วย foreign key เป็นสะพานเชื่อมระหว่างคนใน token กับแถวในตารางเรา
ใช้ตอนไหน ทุกครั้งที่ join ตาราง ตอน login — เอา sub จาก token มาค้นว่า WHERE uid = ?
ถ้าย้าย identity provider ไม่กระทบ ข้อมูลทุกตารางยังชี้ถูก ต้อง map ใหม่ เพราะ sub ของเจ้าใหม่เป็นคนละค่า

แถวสุดท้ายคือเหตุผลจริง ๆ ที่ต้องแยกสองคอลัมน์
ลองคิดตามว่าถ้าเราขี้เกียจ แล้วเอา sub ของ Keycloak มาเป็น primary key ตรง ๆ เลย วันหนึ่งบริษัทเปลี่ยนไปใช้ Google Workspace หรือย้าย Keycloak ไปอีก realm — sub ของทุกคนเปลี่ยนหมด แปลว่า primary key เปลี่ยนหมด แปลว่าทุก foreign key ในทุกตารางที่ชี้มาหา user พังหมดพร้อมกัน
พอแยกเป็นสองคอลัมน์ วันนั้นเราแค่ UPDATE users SET uid = ... — ข้อมูลอื่นไม่รู้เรื่องด้วยเลย

นี่เป็นรูปแบบที่มีชื่อเรียกว่า internal ID กับ external ID และใช้กับทุกอย่างที่มาจากข้างนอก ไม่ใช่แค่ผู้ใช้ — เช่น stripe_customer_id หรือ line_user_id ก็ตรรกะเดียวกัน

🎒 เทียบให้เห็นภาพ — เลขประจำตัวนักเรียน กับ เลขบัตรประชาชน

โรงเรียนออกเลขประจำตัวนักเรียนให้คุณเอง แล้วใช้เลขนี้ในเอกสารภายในทุกใบ — ใบเช็คชื่อ ใบเกรด ใบยืมหนังสือห้องสมุด ทุกใบอ้างเลขนี้
ส่วนเลขบัตรประชาชน โรงเรียนไม่ได้เป็นคนตั้ง มันมาจากข้างนอก โรงเรียนแค่จดไว้ในทะเบียนช่องหนึ่ง เพื่อจะได้จับคู่คุณกับตัวตนที่ออกมาจากข้างนอกได้ถูกคน

id = เลขประจำตัวนักเรียน — แอปเราตั้งเอง และทุกตาราง (recipes, ratings) ชี้มาที่เลขนี้
uid = เลขบัตรประชาชน — Keycloak เป็นคนออก (ก็คือ claim sub) เราแค่จดไว้เพื่อใช้ตอน login ว่า "คนที่ถือ token ใบนี้ คือแถวไหนในตารางเรา"

ที่ไม่เหมือนกัน: เลขบัตรประชาชนไทยติดตัวไปทั้งชีวิต ย้ายโรงเรียนกี่ครั้งก็เลขเดิม — แต่ sub ของ Keycloak ผูกอยู่กับ realm และ provider ตัวนั้น วันที่ย้ายไป identity provider เจ้าใหม่ ทุกคนได้เลขใหม่หมด
ตรงนี้แหละคือเหตุผลที่ห้ามเอา sub มาเป็น primary key ตรง ๆ — ไม่ใช่เพราะมันสั้นไปยาวไป แต่เพราะมันเป็นเลขที่คนอื่นเป็นเจ้าของ และเปลี่ยนได้โดยที่เราไม่ได้สั่ง

🗣 ศัพท์ที่เพิ่งเรียนไป
primary key
คอลัมน์ที่ใช้ระบุแถวในตารางนี้แบบไม่ซ้ำ · ในตารางเราคือ id · ประโยคที่ใช้ได้เลย: "primary key ของตารางนี้เป็น UUID ที่เราสร้างเองนะ ไม่ได้เอามาจาก Keycloak"
foreign key
คอลัมน์ในตารางหนึ่งที่ชี้ไปหา primary key ของอีกตาราง · ประโยคที่ใช้ได้เลย: "recipes.user_id เป็น foreign key ไปหา users.id นะ ไม่ใช่ไปหา uid" — พูดแบบนี้อีกฝ่ายจะเห็นภาพทันทีว่าทำไมสองคอลัมน์นี้ห้ามสลับกัน
internal ID / external ID
ชื่อเรียกรูปแบบที่เพิ่งใช้ไป: เก็บรหัสจากระบบข้างนอกไว้คนละคอลัมน์กับ primary key ของเราเอง · พิมพ์ถาม AI ว่า "internal id vs external id pattern with an external identity provider" จะเจอทั้งเหตุผลและตัวอย่าง schema
identity provider (IdP)
ระบบที่รับหน้าที่ยืนยันตัวตนแทนแอปเรา — Keycloak, Google, LINE เป็น IdP ทั้งนั้น · ประโยคที่ใช้ได้เลย: "ถ้าวันหนึ่งเราย้าย IdP ตารางไหนต้อง map ใหม่บ้างครับ"
claim
ช่องข้อมูลหนึ่งช่องใน JWT — sub, email, preferred_username แต่ละตัวเรียกว่า claim หนึ่งตัว · ประโยคที่ใช้ได้เลย: "token ที่ Keycloak ออกให้ มี claim email ติดมาด้วยไหมครับ" (หัวข้อ 9 จะบอกว่าคำตอบขึ้นกับ scope ที่ขอไป)

แล้วทำไม uid ต้อง UNIQUE

ตอน login โค้ดจะทำสิ่งนี้: เอา sub จาก id_token ไปค้นว่า "คนนี้เคยมาแล้วหรือยัง" ด้วย WHERE uid = ?
ถ้าคำค้นนี้คืนมาสองแถว โค้ดจะทำอะไรถูก? แถวไหนคือ "เขา"? สูตรอาหารที่เขาเขียนไว้ผูกกับแถวไหน?
UNIQUE คือการบังคับให้คำถามนี้มีคำตอบเดียวเสมอ ในระดับฐานข้อมูล — ไม่ได้ฝากไว้กับความระวังของโค้ด

และมันสำคัญเพราะ race condition — สถานการณ์ที่สองคำขอวิ่งพร้อมกันแล้วเหยียบกัน
ผู้ใช้กดปุ่ม login รัว ๆ สองครั้ง เกิดสอง request พร้อมกัน ทั้งคู่ค้นแล้วไม่เจอ (เพราะยังไม่มีใครเขียน) ทั้งคู่จึงสั่ง INSERT
ถ้าไม่มี UNIQUE คุณได้ผู้ใช้ซ้ำสองแถวเงียบ ๆ · ถ้ามี ฐานข้อมูลจะปฏิเสธแถวที่สอง แล้วโค้ดค่อยจับ error นั้นแล้วไปอ่านแถวที่มีอยู่แทน
index ยังทำให้การค้นเร็วขึ้นด้วย เพราะทุก request ที่มี token ต้องค้นด้วย uid — ถ้าไม่มี index ฐานข้อมูลต้องไล่อ่านทั้งตารางทุกครั้ง

8 · เรื่องที่สไลด์ไม่ได้พูด — UNIQUE ชนกับ soft delete

ก่อนอื่นนิยามศัพท์: soft delete คือการ "ลบ" ที่ไม่ได้ลบจริง — แทนที่จะ DELETE แถวทิ้ง เราแค่ประทับเวลาลงคอลัมน์ deleted_at แล้วให้ทุก query กรองแถวที่มีค่าออกไป
ข้อดีคือกู้คืนได้ และข้อมูลที่ผูกอยู่ (สูตรอาหารของคนนี้) ไม่หายตามไปด้วย
คอลัมน์ deleted_at TIMESTAMPTZ กับ index idx_users_deleted_at ในสไลด์ คือรูปแบบที่ GORM รู้จักโดยตรง ผมยืนยันด้วยการรันจริง: 

// field ที่ต้องประกาศฝั่ง GORM — ชนิดต้องเป็น gorm.DeletedAt เท่านั้น ไม่ใช่ *time.Time
DeletedAt gorm.DeletedAt `gorm:"index:idx_users_deleted_at"`
1) Create OK, id = bae1dc16-8585-4f91-bbe6-d30160e7dbdd
2) db.Delete() -> DeletedAt.Valid = true
3) Count() ปกติ (ถูกกรอง deleted_at IS NULL) = 0
4) Unscoped().Count() (นับแถวที่ soft delete ด้วย) = 1

บรรทัด 3 กับ 4 คือหัวใจ: Count() ธรรมดาได้ 0 (มองไม่เห็นแล้ว) แต่ Unscoped().Count() ได้ 1 — แถวยังอยู่ในตารางจริง ๆ GORM แค่แอบเติม WHERE deleted_at IS NULL ให้ทุก query

ทายก่อนรัน — ข้อนี้คือข้อที่สไลด์ข้ามไป และมันจะกัดคุณตอน production

ลำดับเหตุการณ์สามขั้น ทุกขั้นเป็นเรื่องปกติที่เกิดได้จริง:

1. ผู้ใช้ sub = "oidc-sub-123" login ครั้งแรก  -> INSERT แถวใหม่ uid = 'oidc-sub-123'
2. ผู้ใช้กด "ลบบัญชี"                          -> UPDATE deleted_at = now()   (soft delete)
3. ผู้ใช้คนเดิม เปลี่ยนใจ login กลับเข้ามาใหม่     -> ???

ขั้นที่ 3 โค้ดจะค้นด้วย WHERE uid = 'oidc-sub-123' ก่อน — ค้นเจอไหม?
แล้วถ้าค้นไม่เจอ มันจะ INSERT แถวใหม่ — INSERT ผ่านไหม?
(ใบ้: sub ของ Keycloak ไม่เคยเปลี่ยนตลอดชีพของ user คนนั้น · และ index idx_users_uid เป็น UNIQUE ธรรมดา ไม่มีเงื่อนไขอะไรต่อท้าย)

ค้นไม่เจอ และ INSERT ไม่ผ่าน — ผู้ใช้คนนี้ติดตาย กลับเข้ามาไม่ได้อีกเลย

SQL ล้วน ๆ ก่อน เพื่อให้เห็นว่าไม่ใช่ความผิดของ ORM:

INSERT 0 1                                # ขั้น 1 สมัครครั้งแรก
UPDATE 1                                  # ขั้น 2 soft delete
     uid      | soft_deleted
--------------+--------------
 oidc-sub-123 | t                         # แถวยังอยู่ในตาราง แค่ถูกทำเครื่องหมายไว้
(1 row)

# ขั้น 3 สมัครกลับเข้ามาด้วย uid เดิม
ERROR:  duplicate key value violates unique constraint "idx_users_uid"
DETAIL:  Key (uid)=(oidc-sub-123) already exists.

ฝั่ง GORM ก็เหมือนกัน และตัวที่โค้ด OIDC ชอบใช้ที่สุด (FirstOrCreate) เจ็บกว่าเพื่อน:

5) สมัครใหม่ uid เดิม -> err = ERROR: duplicate key value violates unique constraint "idx_users_uid" (SQLSTATE 23505)
6) FirstOrCreate        -> err = ERROR: duplicate key value violates unique constraint "idx_users_uid" (SQLSTATE 23505)

FirstOrCreate คือ "หาก่อน ถ้าไม่เจอค่อยสร้าง" — ฟังดูปลอดภัยที่สุด แต่กลับเป็นกับดักพอดี:
ท่อน First ถูกเงื่อนไข deleted_at IS NULL ที่ GORM แอบเติมให้ทุก query กรองออก เลยหาไม่เจอ
(GORM เรียกตัวกรองอัตโนมัตินี้ว่า scope — คนละความหมายกับ scope ของ OAuth ในหัวข้อ 3 ที่แปลว่า "ขอสิทธิ์อะไรบ้าง" ชื่อพ้องกันเฉย ๆ)
ท่อน Create จึงทำงาน แล้วไปชน unique ของแถวที่ถูกลบ ซึ่งยังนอนอยู่ในตาราง
ผลคือทางตัน: หาก็ไม่เจอ สร้างก็ไม่ได้ — และมันเกิดกับผู้ใช้ที่เคยลบบัญชีทุกคน

ทางแก้: partial unique index — บังคับ unique เฉพาะแถวที่ยังไม่ถูกลบ

PostgreSQL ให้เติม WHERE ท้าย index ได้ แปลว่า "กฎข้อนี้ใช้กับแถวที่เข้าเงื่อนไขนี้เท่านั้น" 

DROP INDEX idx_users_uid;
CREATE UNIQUE INDEX idx_users_uid ON users (uid) WHERE deleted_at IS NULL;

ผลรันหลังแก้ — ลอง INSERT ซ้ำสองรอบติดกัน:

DROP INDEX
CREATE INDEX
INSERT 0 1                               # รอบแรก: สมัครใหม่หลัง soft delete → ผ่านแล้ว
     uid      | active
--------------+--------
 oidc-sub-123 | f                          # แถวเก่าที่ถูกลบ
 oidc-sub-123 | t                          # แถวใหม่ที่ยัง active
(2 rows)

ERROR:  duplicate key value violates unique constraint "idx_users_uid"
DETAIL:  Key (uid)=(oidc-sub-123) already exists.   # รอบสอง: ยังกัน uid ซ้ำในแถวที่ active อยู่

ได้ทั้งสองอย่างพร้อมกัน: สมัครกลับเข้ามาได้ และ ยังกันซ้ำในหมู่คนที่ยัง active
ฝั่ง GORM เขียนเป็น tag แบบนี้: gorm:"uniqueIndex:idx_users_uid,where:deleted_at IS NULL"
ถ้าจะแก้ในโปรเจกต์จริง อย่าไปแก้ไฟล์ migration เดิมที่รันไปแล้ว — ให้สร้าง migration ใหม่ที่ DROP INDEX แล้ว CREATE ใหม่ (คนอื่นในทีมรันไฟล์เดิมไปแล้ว แก้ไฟล์เดิมจะทำให้ฐานของแต่ละคนไม่เหมือนกัน)

🧾 เทียบให้เห็นภาพ — กลับมาสมัครสมาชิกร้านด้วยเบอร์เดิม

คุณเคยสมัครสมาชิกร้านไว้ด้วยเบอร์มือถือ วันหนึ่งขอยกเลิกสมาชิก พนักงานไม่ได้ลบประวัติทิ้ง เขาแค่กดปุ่มให้บัญชีนั้นเป็น "ยกเลิกแล้ว" เพราะประวัติการซื้อยังต้องเก็บไว้
วันหลังคุณเปลี่ยนใจ กลับมาสมัครใหม่ด้วยเบอร์เดิม — แล้วเจอสองด่านนี้:
· พนักงานค้นเบอร์คุณในหน้าจอ หาไม่เจอ เพราะหน้าค้นหากรองบัญชีที่ยกเลิกออกไปหมดแล้ว
· พนักงานเลยกดสมัครใหม่ให้ ระบบเด้งว่า "เบอร์นี้มีสมาชิกอยู่แล้ว" เพราะกฎห้ามเบอร์ซ้ำยังนับรวมบัญชีที่ยกเลิกด้วย
หาก็ไม่เจอ สมัครก็ไม่ได้ — นี่คือเหตุการณ์เดียวกับ FirstOrCreate เป๊ะ ๆ: ท่อน First โดน deleted_at IS NULL กรองออก · ท่อน Create ไปชน idx_users_uid ของแถวที่ถูกลบซึ่งยังนอนอยู่ในตาราง

ที่ไม่เหมือนกัน: ที่ร้านคุณเรียกผู้จัดการมาปลดล็อกให้ได้ในนาทีนั้น แต่ unique index ไม่มีใครมาปลดให้ตอนโปรแกรมกำลังรัน มันเป็นกฎที่ฝังอยู่ในตัวฐานข้อมูล
ทางออกจึงมีแค่สองทาง และทั้งคู่ต้องลงมือเขียนเอง: แก้ตัวกฎ (ทำให้เป็น partial index อย่างที่กล่องข้างบนทำ) หรือ เขียนโค้ดไปปลดแถวเก่าเอง ด้วย Unscoped() ของ GORM แล้วเคลียร์ deleted_at ทิ้ง
อีกจุดที่การเปรียบเทียบพาไปผิดได้: ที่ร้านคุณอาจอยากได้ประวัติเก่ากลับมาด้วย แต่ท่า partial index ในกล่องข้างบนไม่ได้กู้แถวเก่าคืน — มันสร้างแถวใหม่คนละ id ของเก่ายังนอนอยู่แบบถูกลบเหมือนเดิม (ผลรันโชว์ 2 แถว)

🗣 ศัพท์ที่เพิ่งเรียนไป
soft delete
"ลบ" ด้วยการประทับเวลาลง deleted_at แล้วให้ทุก query กรองออก แทนที่จะ DELETE จริง · ประโยคที่ใช้ได้เลย: "ตารางนี้เป็น soft delete นะ แถวยังอยู่ อย่าลืมตอนเขียน query"
partial index
index ที่มี WHERE ต่อท้าย แปลว่า "กฎข้อนี้ใช้กับแถวที่เข้าเงื่อนไขเท่านั้น" · พิมพ์ถาม AI ว่า "postgres partial unique index soft delete deleted_at" จะได้ท่าเดียวกับที่ใช้แก้ในกล่องข้างบน
unique violation · SQLSTATE 23505
รหัสมาตรฐานที่แปลว่า "แถวนี้ชน unique" · โผล่ในผลรันทั้ง SQL ล้วนและฝั่ง GORM · รหัสนี้เอาไปเทียบในโค้ดได้ตรงกว่าการอ่านข้อความ error เป็นสตริง เพราะข้อความเปลี่ยนตามภาษาและเวอร์ชันได้
upsert · ON CONFLICT
"มีอยู่แล้วก็อัปเดต ยังไม่มีก็แทรกใหม่" จบในคำสั่งเดียว ไม่ต้อง SELECT ก่อน · เป็นท่าที่โค้ด handler ท้ายหัวข้อ 9 ใช้แทน FirstOrCreate · ประโยคที่ใช้ได้เลย: "ตรงนี้ทำเป็น upsert on uid ดีกว่าไหมครับ จะได้ไม่มีช่องให้ race"
Unscoped()
คำสั่งของ GORM ที่บอกว่า "รอบนี้อย่าเติมตัวกรอง deleted_at IS NULL ให้" · ใช้ตอนอยากเห็นหรืออยากปลดแถวที่ถูก soft delete — ไม่ใส่ก็มองไม่เห็นแถวนั้นเลย

บั๊กเงียบอีกตัวจากบรรทัด id UUID PRIMARY KEY

ย้อนกลับไปดู \d users ในหัวข้อ 6: คอลัมน์ id ไม่มี DEFAULT แปลว่าฐานข้อมูลไม่ได้สร้าง UUID ให้ — โค้ดฝั่งเราต้องเซ็ตเอง
ผมลองรัน GORM แบบที่ลืมเซ็ต ID ดูสองรอบ (คนละ uid): 

รอบแรก:
Create โดยไม่เซ็ต ID -> err=<nil>, id=00000000-0000-0000-0000-000000000000

รอบสอง (uid ต่างกัน):
Create โดยไม่เซ็ต ID -> err=ERROR: duplicate key value violates unique constraint "users_pkey" (SQLSTATE 23505)

แถวในตาราง:
                  id                  |    uid
--------------------------------------+-----------
 00000000-0000-0000-0000-000000000000 | no-id-set
(1 row)
⚠️ ผู้ใช้คนแรกเข้าได้ คนที่สองถึงจะพัง

Go ให้ค่าเริ่มต้นของ UUID เป็นศูนย์ทั้งหมด 00000000-…-000000000000 ซึ่งเป็นค่าที่ถูกต้องตามชนิด uuid — ฐานข้อมูลจึงรับไว้เฉย ๆ ไม่บ่นอะไร
บั๊กนี้เลยผ่านทุกการทดสอบที่มี user คนเดียว แล้วไประเบิดตอนมีคนที่สองสมัคร ด้วย error ที่พูดถึง users_pkey ซึ่งไม่ได้ชี้ไปที่ต้นเหตุเลย
แก้ได้สองทาง เลือกทางใดทางหนึ่ง:
· ฝั่งฐานข้อมูล — เติม DEFAULT gen_random_uuid() ให้คอลัมน์ id ใน migration (ฟังก์ชันนี้มีมาให้ใน PostgreSQL 13 ขึ้นไป ไม่ต้องลง extension)
· ฝั่ง Go — เขียน BeforeCreate hook ใน GORM ให้สร้าง UUID ก่อนบันทึกทุกครั้ง

9 · last_signed_in_at และ preferred_username มาจากไหน

สองคอลัมน์นี้ในสไลด์ไม่มีคำอธิบาย แต่มันไม่ใช่ของที่เราคิดขึ้นเอง — มันมาจากข้อมูลใน token ที่ Keycloak ส่งมา
นี่คือ payload จริงที่ผมถอดออกมาจาก token ของ Keycloak (ตัดเฉพาะบรรทัดที่เกี่ยว): 

{
  "sub": "5da7f595-196f-44a6-8dfd-64306ce734d6",      // → คอลัมน์ uid
  "preferred_username": "chef1",                     // → คอลัมน์ preferred_username
  "name": "Chef One",                                // → คอลัมน์ name
  "email": "chef1@wongnok.dev",                      // → คอลัมน์ email
  "given_name": "Chef",
  "family_name": "One",
  "email_verified": true,
  "iat": 1787368445,
  "exp": 1787368745                              // = iat + 300 วินาที
}
คอลัมน์มาจากไหนเขียนตอนไหน
uidclaim subตอน login ครั้งแรก แล้วไม่แก้อีก
email, nameclaim email, nameอัปเดตทุกครั้งที่ login (ผู้ใช้อาจไปแก้ที่ Keycloak)
preferred_usernameclaim preferred_usernameเหมือนกัน — เอาไว้โชว์เป็นชื่อผู้ใช้ในแอป
last_signed_in_atไม่มีใน claim — เราเซ็ตเป็น now() เองทุกครั้งที่ callback สำเร็จ
bioไม่มีใน claim — ผู้ใช้พิมพ์เองในแอปเราตอนผู้ใช้แก้โปรไฟล์
image_urlไม่มีใน claim — ผู้ใช้ใส่ลิงก์รูปเองในแอปเราตอนผู้ใช้แก้โปรไฟล์ (สไลด์ Post เพิ่งเพิ่มคอลัมน์นี้มา)

สามแถวล่างคือคำตอบของคำถามในหัวข้อ 1 ว่า "ทำไมต้องมีตาราง users ในเมื่อ Keycloak เก็บให้แล้ว" — bio · image_url · last_signed_in_at เป็นของแอปเรา Keycloak ไม่มีให้และไม่ควรมี

⚠️ claim ที่คุณคิดว่ามีอยู่แล้ว อาจไม่มา ถ้าไม่ได้ขอ

email กับ preferred_username จะมาก็ต่อเมื่อขอ scope profile และ email ซึ่งเป็นเหตุผลที่โค้ดในหัวข้อ 3 เขียนว่า Scopes: []string{oidc.ScopeOpenID, "profile", "email"}
(ค่าเริ่มต้นของ Keycloak ใส่ scope สองตัวนี้ให้อยู่แล้ว แต่ถ้าใครไปปรับ client scope แล้วโค้ดไม่ได้ขอเอง คุณจะได้ token ที่ไม่มีอีเมล แล้วบันทึกช่องว่างลงฐาน)
อีกเรื่องที่ต้องรู้จาก payload ข้างบน: exp ห่างจาก iat แค่ 300 วินาที — access token ของ Keycloak 26.7 อายุ 5 นาที ไม่ใช่ 1 ชั่วโมงอย่างที่หลายคนจำ (refresh token 1800 วินาที = 30 นาที)

type User struct {                                 // map ตรงกับตารางในหัวข้อ 6
	ID                string `gorm:"type:uuid;primaryKey"`
	Email             string
	Name              string
	Bio               string
	UID               string `gorm:"column:uid"`
	PreferredUsername string
	LastSignedInAt    *time.Time
	CreatedAt         time.Time
	UpdatedAt         time.Time
	DeletedAt         gorm.DeletedAt                // ← ตัวนี้แหละที่เปิดโหมด soft delete
}

type Server struct {
	Auth *auth.Authenticator
	DB   *gorm.DB
}

// GET /auth/login — สุ่ม state ใหม่ทุกครั้ง เก็บใส่ cookie แล้วค่อย redirect
func (s *Server) Login(w http.ResponseWriter, r *http.Request) {
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		http.Error(w, "cannot generate state", http.StatusInternalServerError)
		return
	}
	state := base64.RawURLEncoding.EncodeToString(b)

	http.SetCookie(w, &http.Cookie{
		Name: "oauth_state", Value: state,
		Path: "/", HttpOnly: true, MaxAge: 300,
	})
	http.Redirect(w, r, s.Auth.LoginURL(state), http.StatusFound)
}

// GET /auth/callback — เทียบ state ก่อน แล้วค่อยเอา code ไปแลก token
func (s *Server) Callback(w http.ResponseWriter, r *http.Request) {
	saved, err := r.Cookie("oauth_state")
	if err != nil || saved.Value != r.URL.Query().Get("state") {
		http.Error(w, "state mismatch", http.StatusBadRequest)
		return
	}

	c, err := s.Auth.Callback(r.Context(), r.URL.Query().Get("code"))
	if err != nil {
		http.Error(w, err.Error(), http.StatusUnauthorized)
		return
	}

	now := time.Now()
	u := User{
		ID:                uuid.NewString(),      // ← หัวข้อ 8 บอกไว้: id ไม่มี DEFAULT เราต้องเซ็ตเอง
		UID:               c.Sub,               // ← claim sub ลงคอลัมน์ uid ตรงนี้
		Email:             c.Email,
		Name:              c.Name,
		PreferredUsername: c.PreferredUsername,
		LastSignedInAt:    &now,                // ← ไม่มีใน claim เราเซ็ตเอง
	}

	// upsert: มี uid นี้อยู่แล้ว = อัปเดต · ยังไม่มี = INSERT ใหม่
	if err := s.DB.Clauses(clause.OnConflict{
		Columns:     []clause.Column{{Name: "uid"}},
		TargetWhere: clause.Where{Exprs: []clause.Expression{clause.Expr{SQL: "deleted_at IS NULL"}}},
		DoUpdates: clause.AssignmentColumns([]string{
			"email", "name", "preferred_username", "last_signed_in_at", "updated_at",
		}),
	}).Create(&u).Error; err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	http.Redirect(w, r, "/", http.StatusFound)
}

# Prompt

สร้างไฟล์ migrations ตาม docs/ARCHITECTURE.md โดยให้ทำเพียง
สร้างไฟล์อย่างเดียวเท่านั้นเพื่อนำไปรีวิวก่อน ไม่ต้องตรวจสอบและ migrate จริง

prompt นี้สั้นแต่มีของอยู่สามชิ้นที่วางไว้ตั้งใจ และเป็นเหตุผลที่มันใช้ได้จริง:

ท่อนในคำสั่งทำหน้าที่อะไร
"ตาม docs/ARCHITECTURE.md" ชี้แหล่งความจริงให้ชัด — ไม่ให้ agent เดาโครงตารางเอาเอง ทุกคอลัมน์ต้องอ้างกลับไปที่เอกสารได้
"ทำเพียงสร้างไฟล์อย่างเดียว" จำกัดขอบเขต — ไม่ให้ไปแก้ไฟล์อื่น ไม่ให้ไปรันคำสั่งอื่น
"ไม่ต้อง migrate จริง" กันไม่ให้แตะฐานข้อมูล — ผลลัพธ์เป็นแค่ไฟล์ที่กลับคืนได้ ไม่ใช่ตารางที่สร้างไปแล้ว

ท่อนที่สามคือท่อนสำคัญที่สุด และเชื่อมกลับมาที่ทั้งบทนี้พอดี: ทุกกับดักในหัวข้อ 8 เป็นสิ่งที่คนรีวิวเห็นได้ตอนอ่านไฟล์ แต่มองไม่เห็นถ้าปล่อยให้รันไปแล้ว
ทั้ง unique index ที่ชนกับ soft delete และ id UUID ที่ไม่มี default — ทั้งคู่ goose up ผ่านฉลุยทั้งคู่ ไม่มี error ให้เห็น
เกณฑ์รีวิว migration ที่ AI เขียนให้ ขั้นต่ำสามข้อ — ทั้งสามข้อมาจากสิ่งที่รันเจอในบทนี้:

  1. ทุก UNIQUE INDEX บนตารางที่มี deleted_at — มี WHERE deleted_at IS NULL ต่อท้ายหรือยัง
  2. ทุก PRIMARY KEY ชนิด UUID — มี DEFAULT ไหม ถ้าไม่มี ใครเป็นคนเซ็ตค่า
  3. ส่วน -- +goose Down — เขียนย้อนกลับไว้ครบจริงไหม หรือปล่อยว่างไว้ (ซึ่งแปลว่าถอยไม่ได้)

บทเรียน 54 จะพูดถึงเรื่องใหญ่กว่านั้น คือโหมดการทำงานร่วมกับ AI agent ว่าจะสั่งงานแบบไหน ควบคุมขอบเขตยังไง ซึ่ง prompt หน้า 54 เป็นตัวอย่างหนึ่งของโหมดเหล่านั้น

1. ทำไมต้องลงทั้ง golang.org/x/oauth2 และ github.com/coreos/go-oidc/v3 ตัวเดียวไม่พอเหรอ?

สองตัวนี้ทำคนละครึ่งของกระบวนการและไม่ทำงานแทนกัน
oauth2 รับผิดชอบขั้นตอนแลกของ: AuthCodeURL ประกอบ URL หน้า login · Exchange ยิง POST ไป token_endpoint เอา code แลกเป็น token
go-oidc รับผิดชอบขั้นตอนตรวจของ: NewProvider อ่าน discovery document · Verifier(...).Verify โหลด public key จาก jwks_uri มาตรวจลายเซ็น แล้วเช็ค iss/aud/วันหมดอายุให้
ตัวเลือกที่ 4 ผิดคนละทาง: มันไม่ใช่ dependency บังคับของกันและกัน — go list -m all หลัง go mod tidy แสดงว่าทั้งคู่เป็น direct dependency ทั้งสองตัว และตัวที่เป็น // indirect จริง ๆ คือ github.com/go-jose/go-jose/v4 v4.1.4 ซึ่งเป็นตัวที่ go-oidc ใช้ทำงานเข้ารหัส
หลักฐานว่าทั้งสองตัวจำเป็น: ผลรัน go run ./cmd/probe มีทั้ง AuthCodeURL (งานของ oauth2) และ error oidc: malformed jwt: … (งานของ go-oidc) อยู่ในโปรแกรมเดียวกัน

2. ใช้ DSN ตามสไลด์เป๊ะ ๆ (postgres://postgres:Pe@devp00l@localhost:5432?database=wongnok) แล้วสั่ง goose up — เกิดอะไรขึ้นจริง ๆ?

ข้อนี้เป็นข้อที่ผมทายผิดเองตอนแรก ผลรันจริงคือฝั่ง Go ผ่านหมด: goose up พิมพ์ OK 20260822031131_create_users.sql (12.99ms) และ psql -d wongnok เห็นตาราง users กับ goose_db_version ส่วนฐาน postgres ขึ้นว่า Did not find any relations.
สาเหตุมีสองข้อ: net/url ของ Go ตัดที่ @ ตัวสุดท้าย รหัสผ่านเลยได้ "Pe@devp00l" ถูกต้อง · และ pgx รู้จักคีย์ database อยู่แล้ว (เป็นชื่อที่มันใช้ภายใน ส่วน dbname เป็นชื่อพ้อง)
แต่พอเอา DSN เดิมไป psql พังทั้งสองจุด: could not translate host name "devp00l@localhost" เพราะ libpq ตัดที่ @ ตัวแรก ตรงข้ามกับ Go · และ invalid URI query parameter: "database" เพราะ psql ไม่รู้จักคีย์นั้น
ตัวเลือกที่ 4 คือความเข้าใจผิดที่ต้องแก้: Go กับ libpq ตีความ DSN เดียวกันไม่เหมือนกัน
เขียนให้ปลอดภัยทั้งสองฝั่ง: postgres://postgres:Pe%40devp00l@localhost:5432/wongnok?sslmode=disable

3. ผู้ใช้ลบบัญชี (soft delete — deleted_at ถูกเซ็ต) แล้ววันหลัง login กลับเข้ามาใหม่ด้วยบัญชี Keycloak เดิม โค้ดเรียก FirstOrCreate — เกิดอะไรขึ้น?

ผลรันจริงคือ err = ERROR: duplicate key value violates unique constraint "idx_users_uid" (SQLSTATE 23505) ทั้งกรณี db.Create ธรรมดาและกรณี FirstOrCreate
เหตุผลที่มันเป็นทางตัน: ท่อน First ถูกตัวกรอง soft delete ของ GORM (deleted_at IS NULL) กรองออก (WHERE deleted_at IS NULL) เลยหาไม่เจอ · ท่อน Create จึงทำงาน แล้วไปชน unique index ของแถวที่ยังนอนอยู่ในตาราง — หาก็ไม่เจอ สร้างก็ไม่ได้
ตัวเลือกที่ 3 คือสิ่งที่หลายคนหวังว่า GORM ทำให้ แต่มันไม่ทำ — การ undelete ต้องเขียนเองด้วย Unscoped()
ยืนยันด้วย SQL ล้วนแล้วว่าไม่ใช่ความผิดของ ORM: INSERT 0 1 → UPDATE 1 → ERROR: duplicate key … Key (uid)=(oidc-sub-123) already exists.
ทางแก้คือ partial unique index: CREATE UNIQUE INDEX idx_users_uid ON users (uid) WHERE deleted_at IS NULL; ซึ่งพอรันแล้วได้ INSERT 0 1 (สมัครใหม่ผ่าน) แต่รอบถัดไปยังได้ ERROR: duplicate key อยู่ (ยังกันซ้ำในแถวที่ active) — ได้ทั้งสองอย่างพร้อมกัน
ฝั่ง GORM: gorm:"uniqueIndex:idx_users_uid,where:deleted_at IS NULL"

4. หลัง tok, err := cfg.Exchange(ctx, code) สำเร็จ จะเอา id_token มา verify ต้องทำยังไง?

oauth2.Token ไม่มี field ชื่อ IDToken — ตัวเลือกที่ 3 จึง compile ไม่ผ่านด้วยซ้ำ
เหตุผล: id_token เป็นของมาตรฐาน OIDC ไม่ใช่ของ OAuth2 library oauth2 จึงไม่รู้จักมัน และเก็บ field ที่ไม่รู้จักไว้ในถุงรวมที่หยิบด้วย tok.Extra(ชื่อ) ซึ่งคืน any จึงต้อง .(string) แปลงเอง
ตัวเลือกที่ 1 ผิดและอันตราย เพราะ access_token กับ id_token เป็น JWT คนละใบ คนละหน้าที่ — ใบหนึ่งบอกว่าทำอะไรได้ อีกใบบอกว่าเป็นใคร
ตัวเลือกที่ 4 ผิดหนึ่งขั้น: Verify คืน *oidc.IDToken ซึ่งยังเป็นก้อนที่ตรวจแล้ว แต่ยังไม่ได้แกะเป็น field — ต้อง idToken.Claims(&c) โดยที่ c เป็น struct ที่มี tag json:"preferred_username" ฯลฯ ถึงจะได้ค่ามาเขียนลงตาราง users

    ยืนยันด้วยตาว่า idx_users_uid ขึ้นคำว่า UNIQUE และคอลัมน์ id ช่อง Default ว่างเปล่า
  6. สร้างบั๊กด้วยมือตัวเอง (สำคัญที่สุด): ยิง SQL สามคำสั่งตามลำดับ — INSERT user ที่ uid='oidc-sub-123' · UPDATE users SET deleted_at = now() WHERE uid='oidc-sub-123' · INSERT ซ้ำด้วย uid เดิม
    จดข้อความ error ที่ได้แบบเต็ม ๆ รวมทั้งบรรทัด DETAIL:
  7. แก้ด้วย migration ใหม่ (อย่าแก้ไฟล์เดิม): goose create fix_users_uid_partial_index sql
    ส่วน Up ให้ DROP INDEX แล้ว CREATE UNIQUE INDEX … WHERE deleted_at IS NULL · ส่วน Down ต้องเขียนย้อนกลับให้ครบด้วย
    รันข้อ 6 ซ้ำ — รอบนี้ต้อง INSERT ผ่าน แล้วลอง INSERT อีกรอบให้ได้ error กลับมา (ยืนยันว่ายังกันซ้ำอยู่)
  8. ทดสอบขาถอย: goose down · goose status · goose up · goose status
    แล้วเปิดตาราง goose_db_version ดู
    ตอบให้ได้ว่าเลข id ในตารางนั้นเรียงต่อกันหรือกระโดด และทำไม
  9. ท้าทาย: ปิดช่อง id UUID ที่ไม่มี default — เลือกทางใดทางหนึ่ง (DEFAULT gen_random_uuid() ใน migration หรือ BeforeCreate hook ใน GORM)
    แล้วเขียนคำอธิบายสามบรรทัดว่าทางที่คุณเลือกมีข้อเสียอะไร เทียบกับอีกทาง
    (ใบ้: ถ้ามีคนยิง SQL เข้าฐานตรง ๆ โดยไม่ผ่านแอปของคุณ ทางไหนยังคุ้มครองอยู่?)
