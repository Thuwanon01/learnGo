struct tag ทั้ง 4 ตัว

tagทำอะไรเขียนยังไงไม่ใส่แล้วเป็นยังไง
envบอกว่า field นี้อ่านจาก env ชื่ออะไร`env:"APP_PORT"`field นั้นไม่ถูกอ่านเลย
envDefaultค่าที่ใช้เมื่อไม่มี env ตัวนั้น`env:"APP_PORT" envDefault:"8080"`ได้ zero value ของ type นั้น
envSeparatorตัวคั่นของ slice`env:"ORIGINS" envSeparator:";"`ใช้ , เป็นค่าเริ่มต้น
envPrefixเติมหน้า key ของ struct ซ้อนPrimary DB `envPrefix:"DB_"`struct ซ้อนอ่าน key เปล่า ๆ (เช่น HOST)

envSeparator ใช้ได้กับ slice ของ type อื่นด้วย ไม่ใช่แค่ []string (ทดสอบ []int ผ่าน) · และ envDefault ก็ถูก split ด้วย separator เดียวกัน (SepDefault มาจาก envDefault:"a,b,c" ทั้งที่ไม่ได้ตั้ง env TAGS เลย)

option ท้าย tag — เขียนต่อหลังชื่อ key คั่นด้วย ,

optionเช็คอะไรerror จริงเมื่อไม่ผ่าน
,required มี env ตัวนี้อยู่ไหม — ไม่ได้เช็คว่าว่างหรือเปล่า env: required environment variable "MYVAR2" is not set
,notEmpty ต้องมี และห้ามเป็นค่าว่าง env: environment variable "NE_VAR" should not be empty
,file เอาค่า env เป็น path แล้วอ่านเนื้อไฟล์มาใส่แทน env: could not load content of file "/definitely/not/here.txt" from variable F_VAR: open /definitely/not/here.txt: no such file or directory
,unset parse เสร็จแล้ว ลบ env ตัวนั้นทิ้ง ไม่มี error — เป็นการกระทำ ไม่ใช่การตรวจ

3 · caarlos0/env — กับดัก

① ,required คู่กับ envDefault = required ไม่มีผลเลย

type A struct { Both        string `env:"MYVAR,required" envDefault:"fallback"` }
type B struct { OnlyRequired string `env:"MYVAR2,required"` }
=== ไม่ตั้ง env เลย ===
[A required+envDefault] value="fallback" err=<nil>
[B required only]       value=""         err=env: required environment variable "MYVAR2" is not set

A มี required แต่ได้ default ไปเงียบ ๆ err=<nil> · B ที่มี required เปล่า ๆ error ตามคาด · เหตุผลอยู่ในซอร์ส env.go ฟังก์ชัน getOr (env.go:648-660 ของ v11.4.1):

func getOr(key, defaultValue string, defExists bool, envs map[string]string) (val string, exists, isDefault bool) {
	value, exists := envs[key]
	switch {
	case (!exists || key == "") && defExists:
		return defaultValue, true, true   // ← บรรทัด 651-652 · คืน exists=true
	case exists && value == "" && defExists:
		return defaultValue, true, true   // ← ตั้งเป็นค่าว่างก็ยังได้ default

พอ exists เป็น true เสมอเมื่อมี envDefault เงื่อนไขที่ env.go:613

if fieldParams.Required && !exists && fieldParams.OwnKey != "" {

ก็ไม่มีวันเป็นจริง

② ,notEmpty คู่กับ envDefault ก็โดนกลบเหมือนกันเป๊ะ

type C struct { BothNotEmpty string `env:"MYVAR3,notEmpty" envDefault:"fallback3"` }
=== ไม่ตั้ง env เลย ===
[C notEmpty+envDefault] value="fallback3" err=<nil>
=== ตั้ง MYVAR3 ว่าง (MYVAR3=) ===
[C notEmpty+envDefault] value="fallback3" err=<nil>

ตั้งเป็นค่าว่างก็ยังได้ default (มี case exists && value == "" && defExists ใน getOr อีกอัน) · กฎที่จำง่ายที่สุด: ใส่ envDefault เมื่อไหร่ required และ notEmpty ตายทันที

⚠ กับดักที่อันตรายกว่าข้อ ① อีก — ,required ไม่กันค่าว่าง 
$ MYVAR2= go run .            # ตั้งเป็นสตริงว่าง
[B required only]       value="" err=<nil>

,required เช็คแค่ “มี env ตัวนี้อยู่ไหม” (ซอร์ส env.go:613 ดูแค่ exists ไม่ได้ดูค่า) · ใน docker-compose.yml การเขียน POSTGRES_DSN= ทิ้งไว้แบบไม่ใส่ค่าเกิดขึ้นง่ายมาก และมันจะผ่านฉลุย
อยากกันค่าว่างจริง ต้องใช้ ,notEmpty และต้องไม่มี envDefault

③ แปลง type ไม่ได้ — struct ไม่ได้ว่างทั้งก้อน

$ APP_PORT=abc go run t1.go
cfg=main.Config{AppPort:0, ShutdownTimeout:10000000000, LogLevel:"DEBUG", PostgresDSN:""}
err=env: parse error on field "AppPort" of type "int": strconv.ParseInt: parsing "abc": invalid syntax
AppPort type=int value=0
ShutdownTimeout type=time.Duration value=10s ms=10000

สังเกตให้ดี: field ที่พังได้ zero value ส่วน field อื่นถูกเติมครบตามปกติ (ShutdownTimeout=10s, LogLevel=DEBUG) · ใครลืมเช็ค err จะได้ AppPort=0 ไปใช้ต่อ แล้ว server จะ bind พอร์ตสุ่ม

④ time.Duration ที่ลืมใส่หน่วย

$ APP_SHUTDOWN_TIMEOUT=10 go run t1.go

เขียน 10 คิดว่าเป็นวินาที — ต้องเขียน 10s · มือใหม่พลาดตรงนี้บ่อยที่สุดในบทนี้

4 · godotenv — autoload vs Load vs Overload

วิธีที่ 1 — autoload (สั้นที่สุด)

import _ "github.com/joho/godotenv/autoload"
// จบ — ไม่ต้องเรียกอะไรอีกเลย os.Getenv ได้ค่าจาก .env ทันที
--- run 1: autoload, no shell env ---
APP_NAME="from-dotenv"
PORT="9999"

ทั้ง package autoload มีแค่ 15 บรรทัด และส่วนที่ทำงานจริงคือ:

import "github.com/joho/godotenv"

func init() {
	godotenv.Load()
}

ไม่รับค่า error กลับมาเลย — ทิ้งทั้งดุ้น · จำบรรทัดนี้ไว้ เดี๋ยวจะได้เห็นราคาของมัน

// พิสูจน์ว่าค่ามาถึงก่อน main จริง (ไม่ใช่แค่อ่านซอร์สเดา)
1) package-level var init: GOOD=set-by-dotenv

// internal/config/humanize.go
func Humanize(err error) string {
	var joined interface{ Unwrap() []error }
	errs := []error{err}
	if errors.As(err, &joined) {
		errs = joined.Unwrap()
	}

	lines := []string{"invalid configuration:"}
	for _, e := range errs {
		var ve validator.ValidationErrors
		if errors.As(e, &ve) {
			for _, fe := range ve {
				lines = append(lines, fmt.Sprintf("  - %s: ผิดกฎ %q", fe.Namespace(), fe.Tag()))
			}
			continue
		}
		lines = append(lines, "  - "+e.Error())
	}
	return strings.Join(lines, "\n")
}

// ใช้ตัวเดียวทั้ง package — validator แคชข้อมูล struct ไว้ข้างใน
var validate = validator.New(validator.WithRequiredStructEnabled())

type Config struct {
	App      App
	Logging  Logging
	Database Database
}

func Load() (Config, error) {
	cfg, err := env.ParseAs[Config]()
	if err != nil {
		return Config{}, fmt.Errorf("parse environment: %w", err)
	}

	// ไม่ wrap ด้วย fmt.Errorf — errors.Join จัดบรรทัดไว้แล้ว
	if err := cfg.Validate(); err != nil {
		return Config{}, err
	}

	return cfg, nil
}

// รวม error ของ "ทุก section" มาแสดงทีเดียว
// เพิ่ม Redis? สร้าง redis.go + เติม 1 บรรทัดตรงนี้
func (cfg Config) Validate() error {
	return errors.Join(
		cfg.App.Validate(),
		cfg.Logging.Validate(),
		cfg.Database.Validate(),
	)
}

12 · flow /login /callback /exchange /logout

endpointใครเป็นคนเรียกเรียกยังไงแตะ Redis ยังไงตอบอะไรกลับ
GET /api/v1/auth/login Browser window.location · ห้าม AJAX SET oauth_state:xxx TTL 300 วินาที 302 + Location: ไปหน้า login ของ Keycloak
GET /api/v1/auth/callback Browser (เดินตาม 302 ของ Keycloak) ไม่มีใครสั่ง — เบราว์เซอร์เดินเองตาม redirect GETDEL oauth_state:xxx แล้ว SET auth_ticket:yyy TTL 30 วินาที 302 + Location: ไป FRONTEND_URL/auth/success?ticket=yyy
(FRONTEND_URL = env ของ API เราเอง ไม่ใช่ของ Keycloak — URL ของหน้าเว็บฝั่ง frontend)
POST /api/v1/auth/exchange Frontend JS axios.post (AJAX) · body {ticket} GETDEL auth_ticket:yyy 200 + body {accessToken, refreshToken, bearerType, expiresAt}
POST /api/v1/auth/logout Frontend JS axios.post (AJAX) · body {refreshToken} ไม่แตะเลย 204 No Content (body ว่าง)
คำที่ต้องแยกให้ออกหมายความว่า
window.location = "…"สั่งเบราว์เซอร์ให้ย้ายทั้งหน้าไปที่ URL นั้น — หน้าปัจจุบันถูกทิ้ง แถบที่อยู่เปลี่ยน และ redirect ต่อ ๆ ไปเบราว์เซอร์เดินตามให้เอง
AJAX (axios.get(…))ยิง request เบื้องหลังโดยไม่เปลี่ยนหน้า — JavaScript ได้ response กลับมาเป็นข้อมูลก้อนหนึ่ง หน้าจอไม่ขยับ
