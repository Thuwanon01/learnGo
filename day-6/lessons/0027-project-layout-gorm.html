จนถึงตอนนี้โค้ดของคุณอยู่ในไฟล์เดียว รันแล้วจบ บทนี้คือการก้าวไปสู่โปรเจกต์จริง — หลายไฟล์ หลาย package มี database จริง

1 · internal/ — โฟลเดอร์ที่ compiler ปกป้องให้

myapp/

internal/ ไม่ใช่แค่ธรรมเนียมการตั้งชื่อ — Go บังคับใช้จริงที่ระดับ compiler

✅ ภายใน module เดียวกัน

❌ จาก module อื่น

ใน บทเรียน 13 คุณเห็นแล้วว่าตัวพิมพ์ใหญ่/เล็กควบคุมการเข้าถึงระดับตัวแปร
internal/ คือกลไกเดียวกันแต่ระดับทั้ง package — ให้ export ชื่อเป็นตัวใหญ่ได้อิสระข้างใน โดยที่โลกภายนอกยังแตะไม่ได้
ผลในทางปฏิบัติ: ของที่อยู่ใน internal/ คุณรีแฟกเตอร์ได้ตลอดเวลาโดยไม่ทำใครพัง — ซึ่งเป็นอิสรภาพที่มีค่ามากในโปรเจกต์ที่อยู่ยาว

ทำไมต้องมี dto.go แยกจาก model.go

// model.go — หน้าตาในฐานข้อมูล
type User struct {
    ID           int64
    Name         string
    Email        string
    PasswordHash string     // ← ห้ามหลุดออกไปเด็ดขาด
}

// dto.go — หน้าตาที่ส่งออกไปข้างนอก
type UserResponse struct {
    ID    int64  `json:"id"`
    Name  string `json:"name"`
    Email string `json:"email"`
}

นี่คือคำตอบของปัญหาที่เจอใน บทเรียน 13 — แยก struct ปลอดภัยกว่าการพึ่ง json:"-" เพราะเพิ่ม field ใหม่เข้า model แล้วลืมใส่ tag = ข้อมูลหลุดทันที
ส่วน DTO ต้องตั้งใจเพิ่ม ถึงจะหลุดออกไปได้

4 · Database พื้นฐานที่ต้องรู้

คำคืออะไรตัวอย่าง
Tableตาราง — แถวคือข้อมูล 1 ชิ้น คอลัมน์คือ fieldusers
Primary Keyคอลัมน์ที่ระบุตัวตนแถวได้ไม่ซ้ำid
Foreign Keyคอลัมน์ที่ชี้ไปหา PK ของอีกตารางorders.user_id
SQLภาษาที่ใช้สั่ง databaseSELECT ... WHERE ...
Transactionกลุ่มคำสั่งที่ สำเร็จทั้งหมด หรือไม่สำเร็จเลยโอนเงิน: หักบัญชี A + เพิ่มบัญชี B

เปิด PostgreSQL ด้วย Docker — ไฟล์ compose.yml:

services:
  postgres:
    image: postgres:17
    container_name: devpool-postgres
    restart: always
    environment:
      POSTGRES_PASSWORD: Pe@devp00l
    ports:
      - "5432:5432"
    volumes:
      - ./pgdata:/var/lib/postgresql/data

volumes คือส่วนที่ทำให้ข้อมูลไม่หายเมื่อลบ container — ไม่ใส่บรรทัดนี้ ข้อมูลหายทุกครั้งที่ docker compose down
ดูข้อมูลด้วยเครื่องมือ GUI ได้ที่ TablePlus

5 · database/sql หรือ GORM

database/sql — มากับ Go

rows, err := db.Query(
    `SELECT id, name FROM orders
     WHERE user_id = ?`, userID,
)
if err != nil {
    return err
}
defer rows.Close()

for rows.Next() {
    var id int
    var name string
    rows.Scan(&id, &name)
}

ยาวกว่า แต่เห็นทุก SQL ที่ยิงจริง · ไม่มี dependency · คุมประสิทธิภาพได้เต็มที่

GORM — ORM ยอดนิยมของ Go

var orders []Order

db.Where(
    "user_id = ?", userID,
).Find(&orders)

สั้นกว่ามาก · แปลง struct ให้อัตโนมัติ · แต่มองไม่เห็นว่ายิง SQL อะไรออกไป

ชุมชน Go แบ่งเป็นสองฝ่ายจริง ๆ ในเรื่องนี้ — ต่างจาก Java/PHP ที่ ORM เป็นค่ามาตรฐาน
หลายทีมเลือก database/sql + sqlc หรือ sqlx เพราะอยากเห็น SQL ตรง ๆ
คอร์สสอน GORM ก็เรียน GORM ให้เป็น — แต่รู้ไว้ว่ามีอีกทางที่นิยมพอ ๆ กัน และต้องอ่าน SQL ออกอยู่ดี เพราะเวลา query ช้า คุณจะต้องไปดูว่า ORM แปลออกมาเป็นอะไร

7 · GORM Model และ CRUD

type User struct {
    ID    uint   `gorm:"primaryKey"`
    Name  string
    Email string `gorm:"unique"`
}

struct tag อีกชุดหนึ่ง — คนละตัวกับ json: ที่เรียนในบทเรียน 13 แต่ทำงานแบบเดียวกัน: เป็น metadata ที่ library อ่านตอนรัน
struct หนึ่งตัวมีได้ทั้งสอง tag พร้อมกัน 

user := User{Name: "Somchai", Email: "somchai@example.com"}
db.Create(&user)

fmt.Println(user.ID)

ยังไม่ได้ query อะไรกลับมา — user.ID เป็นอะไร?

ผมรันกับ Postgres จริง ได้:

Create: RowsAffected=1 Error=<nil>  → ID ที่ได้กลับมา = 1

GORM เขียนค่ากลับเข้า struct ของคุณ — เพราะคุณส่ง &user ไป มันจึงแก้ตัวจริงได้ (บทเรียน 07)
ถ้าส่ง user เฉย ๆ ไม่มีดาว → GORM แก้สำเนา → ID ของคุณยังเป็น 0

สังเกตอีกอย่าง: GORM ไม่คืน error เป็นค่าที่สอง
มันคืน *gorm.DB ที่มี .Error และ .RowsAffected อยู่ข้างใน — ต่างจากทุกอย่างที่คุณเรียนมา ตั้งแต่บทเรียน 05
แปลว่า ลืมเช็ค .Error ได้ง่ายมาก เพราะ compiler ไม่บังคับเหมือน multiple return value 

// CREATE
db.Create(&user)

// READ
var user User
db.First(&user, 1)                                  // หา id = 1

var users []User
db.Find(&users)                                    // เอาทั้งหมด

var vipUsers []User
db.Where("name = ?", "Somchai").Find(&vipUsers)  // มีเงื่อนไข

// UPDATE
db.Model(&user).Update("Name", "Somsri")

// DELETE
db.Delete(&user, 1)

ชื่อตารางถูกเดาให้อัตโนมัติ — ผมตรวจแล้ว struct User → ตาราง users (ตัวเล็ก + เติม s) · OrderItem → order_items
ใช้เครื่องหมาย ? ใน Where เสมอ อย่าต่อสตริงเอง — นั่นคือช่องโหว่ SQL injection

8 · สี่กับดักของ GORM ที่ผมเจอตอนทดสอบ

8.1 First ฟ้องว่าไม่เจอ แต่ Find ไม่ฟ้อง

var missing User
err := db.First(&missing, 9999).Error
// err = record not found · errors.Is(err, gorm.ErrRecordNotFound) = true

var none []User
res := db.Where("id = ?", 9999).Find(&none)
// res.Error = <nil>  ·  res.RowsAffected = 0   ← ไม่ error!

⚠️ นี่คือกับดักที่คนเจอบ่อยที่สุดใน GORM
First คาดหวังว่าต้องเจอ → ไม่เจอ = error
Find คาดหวัง "เอาเท่าที่มี" → ไม่มี = slice ว่าง ไม่ใช่ error
ถ้าเช็คแต่ .Error หลัง Find คุณจะไม่มีวันรู้ว่าไม่เจอ — ต้องเช็ค len(users) == 0 หรือ RowsAffected เอง
ใช้ errors.Is(err, gorm.ErrRecordNotFound) เสมอ ไม่ใช่ == (บทเรียน 12)

8.2 Update/Delete ที่ไม่มีเงื่อนไข ถูกบล็อก

db.Model(&User{}).Update("Balance", 0).Error
// WHERE conditions required
// errors.Is(err, gorm.ErrMissingWhereClause) = true

db.Delete(&User{}).Error
// WHERE conditions required

GORM กันไม่ให้คุณลบทั้งตารางโดยบังเอิญ — เป็นฟีเจอร์ ไม่ใช่บั๊ก · ถ้าตั้งใจจริงต้องใส่ .Where("1 = 1") ให้ชัดเจน

8.3 Transaction — rollback ทำงานจริง

err := db.Transaction(func(tx *gorm.DB) error {
    tx.Model(&Account{}).Where("owner = ?", "A").
        Update("balance", gorm.Expr("balance - ?", 50))

    return errors.New("ระบบพังกลางคัน")   // ← คืน error = rollback
})

ผมทดสอบกับยอดเงิน A = 100:

Transaction คืน error → ระบบพังกลางคัน
ยอดของ A หลัง rollback = 100   ← ไม่ถูกหัก

เมื่อ callback คืน nil GORM จะ commit ให้ — ผมลองโอน 50 จาก A ไป B:

หลัง transaction สำเร็จ: [{ID:1 Owner:A Balance:50} {ID:2 Owner:B Balance:150}]
💡 กฎเดียวที่ต้องจำเรื่อง transaction ใน GORM

callback คืน error = rollback · คืน nil = commit
และต้องใช้ tx ข้างใน ไม่ใช่ db — เผลอใช้ db เมื่อไหร่ คำสั่งนั้นจะอยู่นอก transaction แล้ว rollback ไม่ถึง (บั๊กที่หายากมาก เพราะทดสอบตอน happy path จะดูปกติทุกอย่าง)

8.4 Soft delete — ลบแล้วยังอยู่

type Post struct {
    ID        uint `gorm:"primaryKey"`
    Title     string
    DeletedAt gorm.DeletedAt `gorm:"index"`   // ← field นี้เปลี่ยนทุกอย่าง
}

ผมสร้าง 1 แถว แล้วสั่ง Delete แล้วนับ:

หลัง Delete: Find เห็น 0 แถว
แต่ Unscoped เห็น 1 แถว   ← ข้อมูลยังอยู่ในตาราง!

⚠️ มี field ชื่อ DeletedAt แบบ gorm.DeletedAt เมื่อไหร่ Delete จะเปลี่ยนพฤติกรรมทันที — จาก DELETE FROM กลายเป็น UPDATE SET deleted_at = now()
ทุก query หลังจากนั้นจะเติม WHERE deleted_at IS NULL ให้เอง
ข้อดี: กู้ข้อมูลคืนได้ · เก็บประวัติได้
ข้อควรระวัง: คนที่ไม่รู้จะงงว่าทำไม UNIQUE(email) ยังชนอยู่ทั้งที่ "ลบไปแล้ว" — เพราะแถวนั้นยังอยู่จริง ๆ
อยากลบจริงต้องใช้ db.Unscoped().Delete(&post)

ทดสอบความเข้าใจ

1. internal/ ต่างจากโฟลเดอร์ธรรมดายังไง?

ผมทดสอบด้วย 2 module: ภายใน module เดียวกัน import ได้ปกติ · จาก module อื่นได้ use of internal package ... not allowed
ชื่อตัวใหญ่ข้างในยังเป็น exported อยู่ (ตัวเลือกที่ 2 ผิด) — แค่ขอบเขตการมองเห็นถูกจำกัดที่ระดับ module
ผลจริง: ของใน internal/ รีแฟกเตอร์ได้โดยไม่ทำใครพัง

2. ทำไมต้องมี dto.go แยกจาก model.go?

ต่อจากบทเรียน 13: พึ่ง json:"-" แปลว่า "ปลอดภัยตราบใดที่ไม่มีใครลืม"
แยก struct แปลว่า "ต้องตั้งใจเพิ่มถึงจะหลุด" — เพิ่ม PasswordHash เข้า model วันไหน DTO ก็ไม่กระทบ
ปลอดภัยโดยค่าเริ่มต้น ดีกว่าปลอดภัยเพราะจำได้

3. db.Where("id = ?", 9999).Find(&users) เมื่อไม่มีข้อมูลตรงเลย?

ผมรันจริงกับ Postgres: Find(id=9999) Error = <nil> · RowsAffected = 0
Find ไม่ถือว่า "ไม่เจอ" เป็นความผิดพลาด — ต่างจาก First ที่คืน ErrRecordNotFound
ถ้าเช็คแต่ .Error คุณจะเข้าใจผิดว่าสำเร็จ — ต้องดู len() หรือ RowsAffected เอง

4. struct มี DeletedAt gorm.DeletedAt แล้วสั่ง db.Delete(&post)?

ผมทดสอบ: หลัง Delete → Find เห็น 0 แถว แต่ Unscoped().Find เห็น 1 แถว
GORM เปลี่ยนเป็น UPDATE SET deleted_at = now() และเติม WHERE deleted_at IS NULL ให้ทุก query
ระวัง unique constraint — แถวที่ "ลบแล้ว" ยังชนอยู่
ลบจริงต้อง db.Unscoped().Delete(...)

    แล้วลอง FindAll บนตารางว่าง — ได้ error หรือ slice ว่าง?

    ตั้งจำนวน worker เท่าไหร่? (ใบ้: หัวข้อ 5 ของบทเรียน 26 แถวสุดท้าย)

สร้างไม่ได้ — ชน unique constraint
เพราะ soft delete ไม่ได้ลบแถวจริง มันแค่ตั้ง deleted_at แถวเดิมที่มี email นั้นยังอยู่ในตาราง และ constraint ระดับ database ไม่รู้จักแนวคิด "ลบแบบนุ่มนวล" ของ GORM เลย
นี่คือเหตุผลที่ระบบจริงมักทำ unique index แบบมีเงื่อนไข (WHERE deleted_at IS NULL) หรือเปลี่ยนไปใช้ hard delete สำหรับ field ที่ต้อง unique
