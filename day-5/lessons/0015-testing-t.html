บทเรียน 15 · Day 5 · Testing

testing.T: เครื่องมือทั้งหมดที่อยู่ใน t

~16 นาที · ต้องมี: บทเรียน 14, บทเรียน 08 (defer)

t ไม่ใช่แค่ที่ไว้บอกว่า "พัง" — มันคือรีโมตควบคุมการทดสอบ หยุดกลางคัน ข้ามทั้งเทส แตกเป็นเทสย่อย รันขนานกัน เก็บกวาดตอนจบ หรือแม้แต่บอกให้ error ชี้ไปที่บรรทัดที่ถูกต้อง บทนี้ไล่ทีละอย่างพร้อมผลรันจริง

1 · t.Run — แตกเทสใหญ่เป็นเทสย่อย

func TestT_Examples(t *testing.T) {
    t.Run("กลุ่ม_A", func(t *testing.T) {
        t.Run("ย่อย_1", func(t *testing.T) {
            t.Log("path: TestT_Examples/กลุ่ม_A/ย่อย_1")
        })
        t.Run("ย่อย_2", func(t *testing.T) {
            t.Log("path: TestT_Examples/กลุ่ม_A/ย่อย_2")
        })
    })
}
ชื่อ subtest ต่อกันเป็น path คั่นด้วย /

  TestT_Examples
      └── กลุ่ม_A
            ├── ย่อย_1     →  TestT_Examples/กลุ่ม_A/ย่อย_1
            └── ย่อย_2     →  TestT_Examples/กลุ่ม_A/ย่อย_2

path นี้เอาไปใส่ -run ได้ตรง ๆ
  go test -run 'TestT_Examples/กลุ่ม_A/ย่อย_1'
  • subtest แต่ละตัวได้ t ตัวใหม่ — พังตัวหนึ่งไม่ลามตัวอื่น
  • Go แปลงเว้นวรรคในชื่อเป็น _ อัตโนมัติ — ตั้งว่า "ลด 10%" จะแสดงเป็น ลด_10%
  • ซ้อนกี่ชั้นก็ได้ แต่เกิน 2 ชั้นมักอ่านยาก

2 · Error vs Fatal — ต่างกันแค่ "หยุดหรือไม่หยุด"

t.Errorf — ฟ้องแล้วไปต่อ

result := 5
if result != 10 {
    t.Errorf("ได้ %d ไม่เท่ากับ 10", result)
}
t.Log("บรรทัดนี้ยังรันต่อ")

t.Fatalf — ฟ้องแล้วหยุดทันที

result := 5
if result != 10 {
    t.Fatalf("ได้ %d ไม่เท่ากับ 10", result)
}
t.Log("บรรทัดนี้จะไม่รัน")

ผลรันจริงของทั้งสองแบบ:

=== RUN   TestErrorContinues/Error_ทำงานต่อ
    behaviour_test.go:13: ได้ 5 ไม่เท่ากับ 10
    behaviour_test.go:15: บรรทัดนี้ยังรันต่อ         ← มา
--- FAIL: TestErrorContinues/Error_ทำงานต่อ

=== RUN   TestFatalStops/Fatal_หยุดทันที
    behaviour_test.go:23: ได้ 5 ไม่เท่ากับ 10 -> หยุด
--- FAIL: TestFatalStops/Fatal_หยุดทันที                ← ไม่มี Log ตามมา

เลือกยังไง — มีกฎง่าย ๆ ข้อเดียว

ถ้าบรรทัดถัดไปจะพังตามแน่ ๆ ให้ใช้ Fatal · นอกนั้นใช้ Error 

user, err := GetUser("U-101")
if err != nil {
    t.Fatalf("GetUser พัง: %v", err)   // ← Fatal เพราะ user เป็น nil
}
if user.Name != "Alice" {                // ถ้าใช้ Error ข้างบน บรรทัดนี้ panic!
    t.Errorf("Name = %q; want Alice", user.Name)
}
if user.Age != 30 {                       // ← Error เพราะอยากเห็นทุกจุดที่ผิด
    t.Errorf("Age = %d; want 30", user.Age)
}
ข้อจำกัดที่ต้องรู้

t.Fatal ต้องเรียกจาก goroutine ของเทสเอง เท่านั้น — เรียกใน goroutine ที่แตกออกไปแล้ว มันจะหยุด goroutine นั้น แต่ไม่หยุดเทส และผลลัพธ์จะเพี้ยน
ใน goroutine ให้ใช้ t.Error แทน

3 · Log / Logf — เห็นเฉพาะตอน -v

func TestLogOnlyWithV(t *testing.T) {
    t.Log("เริ่มเทส...")
    t.Logf("input = %d, expected = %d", 5, 10)
}

สั่ง go test . เฉย ๆ (ไม่ใส่ -v) จะเห็นข้อความไหม?

ไม่เห็นเลย — ผมรันทดสอบแล้ว ได้แค่ ok

พอใส่ -v ถึงจะโผล่มา: 

=== RUN   TestLogOnlyWithV/Log_ใช้ดีบัก
    behaviour_test.go:40: เริ่มเทส...
    behaviour_test.go:41: input = 5, expected = 10
--- PASS: TestLogOnlyWithV/Log_ใช้ดีบัก

ยกเว้นกรณีเดียว: ถ้าเทสนั้น fail จะโชว์ log ให้เสมอ แม้ไม่ใส่ -v — ออกแบบมาให้ log ที่จำเป็นตอนดีบักโผล่มาเอง
อย่าใช้ fmt.Println ในเทส เพราะมันพิมพ์ตลอดเวลา รกจอ และไม่มีเลขบรรทัดกำกับ

4 · Skip — ข้ามเทสตามเงื่อนไข

func TestHeavyStuff(t *testing.T) {
    if testing.Short() {       // true เมื่อสั่ง go test -short
        t.Skip("ข้ามเพราะอยู่ใน short mode")
    }
    t.Log("ถ้าไม่ใช่ short mode จะรันบรรทัดนี้")
}
คำสั่งผลรันจริง
go test -v .--- PASS: + log ปกติ
go test -v -short .--- SKIP: พร้อมเหตุผล

SKIP ไม่ใช่ FAIL — เทสที่ถูกข้าม ไม่ทำให้ build แดง แต่ยังปรากฏในรายงาน ต่างจากการ comment โค้ดทิ้งซึ่งจะหายไปเลยและไม่มีใครนึกถึงอีก
ใช้ Skip กับ: เทสที่ต้องต่อเน็ต, เทสที่ช้ามาก, เทสที่รันได้เฉพาะบน OS บางตัว (runtime.GOOS)

5 · t.Failed() — ถามว่าตอนนี้พังไปหรือยัง

result := 5
if result != 10 {
    t.Error("ไม่เท่ากับ 10")     // Error ไม่หยุด → มาถึงบรรทัดล่างได้
}

if t.Failed() {
    t.Log("เทสนี้ fail แล้ว จะ dump ข้อมูลเพิ่ม")
}

ประโยชน์จริง: เก็บหลักฐานเฉพาะตอนพัง — เช่น dump ทั้ง struct ออกมา, เซฟ response ลงไฟล์, print SQL ที่รันไป ถ้าทำทุกครั้งจะรกเกินไป แต่ตอนพังคือตอนที่ต้องการมากที่สุด

6 · t.Cleanup — defer ที่ฉลาดกว่า

t.Cleanup(func() { t.Log("cleanup #1 (ลงทะเบียนก่อน)") })
t.Cleanup(func() { t.Log("cleanup #2 (ลงทะเบียนหลัง)") })
t.Log("กำลังเทส...")
t.Error("จงใจ fail")

เทส fail แล้ว cleanup จะรันไหม แล้วรันเรียงยังไง?

รันครบทั้งสองอัน และเรียงแบบ LIFO — ผมรันทดสอบแล้ว:

behaviour_test.go:61: กำลังเทส...
behaviour_test.go:62: จงใจ fail
behaviour_test.go:60: cleanup #2 (ลงทะเบียนหลัง)   ← หลังสุดออกก่อน
behaviour_test.go:59: cleanup #1 (ลงทะเบียนก่อน)
--- FAIL: TestCleanupOrder/Cleanup_ทำความสะอาด

เป็นกฎเดียวกับ defer จาก บทเรียน 08 — last in, first out

แล้วต่างจาก defer ตรงไหน

defert.Cleanup
รันเมื่อฟังก์ชันที่เขียน defer จบเทส (และ subtest ทั้งหมด) จบ
เขียนใน helper แล้วใช้ได้ไหม❌ รันทันทีที่ helper จบ✅ รอจนเทสจบจริง
ลำดับLIFOLIFO
รันตอน fail/panic✅✅
// นี่คือเหตุผลที่ Cleanup มีอยู่ — helper ที่เก็บกวาดตัวเองได้
func newTestDB(t *testing.T) *DB {
    t.Helper()
    db := openTempDB()
    t.Cleanup(func() { db.Close() })   // ← ถ้าใช้ defer จะปิดทันทีที่ออกจาก newTestDB
    return db
}

func TestSomething(t *testing.T) {
    db := newTestDB(t)    // เรียกบรรทัดเดียว ได้ทั้ง setup และ teardown
    // ... ใช้ db ได้ตามปกติ ...
}

7 · t.Helper() — ทำให้ error ชี้ถูกบรรทัด

❌ ไม่ใส่ t.Helper()

func assertEqual(
    t *testing.T, got, want int,
) {
    if got != want {
        t.Errorf("got %d, want %d",
            got, want)
    }
}

✅ ใส่ t.Helper()

func assertEqual(
    t *testing.T, got, want int,
) {
    t.Helper()   // ← บรรทัดเดียว
    if got != want {
        t.Errorf("got %d, want %d",
            got, want)
    }
}

ผลรันจริง — สังเกตเลขบรรทัด:

=== RUN   TestHelperDemonstration/ไม่ใส่_Helper
    parallel2_test.go:43: got 5, want 10      ← ชี้เข้าไปใน helper (ไร้ประโยชน์)
=== RUN   TestHelperDemonstration/ใส่_Helper
    parallel2_test.go:59: got 5, want 10      ← ชี้บรรทัดที่ "เรียก" helper (ถูก!)

เทสหนึ่งไฟล์เรียก assertEqual 30 ครั้ง — ถ้าไม่ใส่ t.Helper() ทุก error จะชี้ไปที่บรรทัดเดียวกันหมด คุณจะไม่รู้เลยว่าเคสไหนพัง
กฎ: helper ที่รับ *testing.T ต้องใส่ t.Helper() เป็นบรรทัดแรกเสมอ

8 · t.Parallel() — และกับดักการวัดเวลาในสไลด์

t.Run(fmt.Sprintf("เคส_%d", n), func(t *testing.T) {
    t.Parallel()                          // ← ต่างที่นี่
    time.Sleep(200 * time.Millisecond)
})

t.Parallel() ทำงานแปลกกว่าที่คิด: มันไม่ได้ "เริ่มรันขนาน" ทันที — มันหยุดเทสย่อยนั้นไว้ก่อน แล้วคืนการควบคุมให้ตัวแม่ พอตัวแม่รันจบทั้งลูปแล้ว Go ถึงปล่อยเทสที่จอดไว้ ทั้งหมดวิ่งพร้อมกัน 

t.Run("PARALLEL", func(t *testing.T) {
    start := time.Now()
    for _, n := range cases {          // 5 เคส เคสละ 200ms
        t.Run(..., func(t *testing.T) {
            t.Parallel()
            time.Sleep(200 * time.Millisecond)
        })
    }
    t.Logf("parallel ใช้เวลา: %v", time.Since(start))
})

บรรทัด Logf จะพิมพ์เวลาประมาณเท่าไหร่?

ประมาณ 140 ไมโครวินาที — ผมรันทดสอบแล้ว ได้ 139.959µs

แต่นั่นไม่ใช่เวลาที่ใช้จริง — มันคือเวลาที่ใช้ "จอดเทสทั้ง 5 ตัวไว้ในคิว" เท่านั้น เพราะ t.Parallel() คืนการควบคุมกลับมาทันที ตอนที่ Logf ทำงาน ยังไม่มีเทสไหนได้รันเลยสักตัว

วิธีวัดให้ถูก — ต้องวัดนอกบล็อกที่สร้าง parallel subtest:

start := time.Now()
t.Run("group", func(t *testing.T) {
    for _, n := range cases {
        t.Run(..., func(t *testing.T) {
            t.Parallel()
            time.Sleep(200 * time.Millisecond)
        })
    }
})
// ถึงบรรทัดนี้ subtest แบบ parallel จบครบแล้ว
t.Logf("ใช้เวลาจริง: %v", time.Since(start))

ผลรันจริงของทั้งสองแบบ:
· sequential (5 × 200ms) → 1.006 วินาที
· parallel วัดถูกวิธี → 201 มิลลิวินาที (เร็วขึ้น 5 เท่าจริง)
· parallel วัดแบบในสไลด์ → 139µs (ตัวเลขหลอกตา)

ก่อนใส่ t.Parallel() ต้องแน่ใจว่า…

เทสไม่แชร์ state กัน — ไม่ใช้ตัวแปร global ร่วมกัน ไม่เขียนไฟล์ชื่อเดียวกัน ไม่แย่ง port เดียวกัน ถ้าแชร์กันจะกลายเป็น flaky test ที่บางทีผ่านบางทีไม่ผ่าน ซึ่งแย่กว่าเทสช้าเยอะ
ใช้ go test -race ช่วยจับ และ -count=10 ช่วยยืนยันว่าเสถียรจริง

9 · ทดสอบความจำ

1. t.Errorf ต่างจาก t.Fatalf ตรงไหน?

ทั้งคู่ทำให้เทส fail เหมือนกัน ต่างที่จะเดินต่อหรือไม่ · กฎเลือก: ถ้าบรรทัดถัดไปจะพังตามแน่ ๆ (เช่นเพิ่งได้ nil มา) ใช้ Fatal นอกนั้นใช้ Error เพื่อจะได้เห็นทุกจุดที่ผิดในรอบเดียว

2. t.Cleanup ดีกว่า defer ตรงไหน?

ทั้งคู่เป็น LIFO และรันตอน fail เหมือนกัน · ต่างที่ ขอบเขต — defer ผูกกับฟังก์ชันที่เขียนมัน ส่วน Cleanup ผูกกับเทส → เขียนใน helper อย่าง newTestDB(t) แล้วมันจะรอปิด db ตอนเทสจบจริง ไม่ใช่ตอนออกจาก helper

3. helper ที่รับ *testing.T ลืมใส่ t.Helper() — ผลคือ?

ทุกอย่างทำงานปกติ แต่เลขบรรทัดใน error ชี้ผิดที่ — ผมรันเทียบแล้ว ไม่ใส่ → ชี้ที่บรรทัด 43 (ในตัว helper) · ใส่ → ชี้ที่บรรทัด 59 (จุดที่เรียก) เรียก helper ตัวเดียว 30 ครั้งแล้วทุก error ชี้ที่เดิม = ดีบักไม่ได้เลย

4. วัด time.Since(start) ในบล็อกที่สร้าง parallel subtest ได้อะไร?

t.Parallel() จอดเทสไว้แล้วคืนการควบคุมทันที → ลูปจบเร็วมาก (ผมวัดได้ 139µs) แล้ว Go ค่อยปล่อยเทสทั้งหมด วิ่งพร้อมกันหลังจากบล็อกแม่จบ วัดให้ถูกต้องวัดนอกบล็อกนั้น — ได้ 201ms เทียบกับ sequential 1.006s

10 · ลงมือทำ

ภารกิจ: ใช้ t ให้ครบทุกเครื่องมือ

ต่อจากโปรเจกต์ calcdemo ในบทที่แล้ว

  1. เขียน TestDiscount ที่มี subtest 3 ตัว: "ลด 10%", "ไม่ลด", "เปอร์เซ็นต์เกิน"
  2. รัน go test -v . แล้วดูว่า ชื่อ subtest ที่แสดง ต่างจากที่คุณตั้งยังไง — จากนั้น -run เจาะเฉพาะตัวกลาง
  3. ในเคส "เปอร์เซ็นต์เกิน" ใช้ t.Fatalf เมื่อ err == nil แล้วอธิบายให้ตัวเองฟังว่าทำไมต้อง Fatal ไม่ใช่ Error
  4. เขียน helper assertNoError(t *testing.T, err error) โดยยังไม่ใส่ t.Helper() → ทำให้เทสพัง → จดเลขบรรทัดใน error ไว้
  5. ใส่ t.Helper() แล้วรันใหม่ → เลขบรรทัดเปลี่ยนเป็นอะไร
  6. เขียน newTempFile(t *testing.T) string ที่สร้างไฟล์ชั่วคราว แล้วลงทะเบียน t.Cleanup ให้ลบไฟล์ทิ้ง — ลองเปลี่ยนเป็น defer ดูว่าพังยังไง
    คำใบ้: os.CreateTemp("", "demo-*.txt")
  7. ท้าทาย 1: ทำการทดลองเรื่อง t.Parallel() ซ้ำ ทั้งแบบวัดผิด (ในบล็อก) และวัดถูก (นอกบล็อก) — ตัวเลขของคุณตรงกับที่ผมได้ไหม (sequential ~1s · parallel ~200ms · วัดผิด ~100µs)
  8. ท้าทาย 2: เขียนเทสที่ใช้ตัวแปร global ร่วมกัน 2 ตัว แล้วใส่ t.Parallel() ทั้งคู่ → รัน go test -race -count=10 . → เห็น WARNING: DATA RACE ไหม?

อ่านต่อจากต้นทาง

แหล่งหลักที่แนะนำ: pkg.go.dev — testing.T เป็นรายการ method ครบทุกตัวพร้อมคำอธิบายสั้น ๆ อ่านผ่านทั้งหน้าครั้งเดียวจะรู้ว่ามีอะไรให้ใช้บ้าง ประมาณ 10 นาที ส่วนเรื่อง t.Parallel() ที่ลึกกว่านี้ (เช่นการใช้ร่วมกับ t.Cleanup) อยู่ในเอกสารของ T.Parallel โดยตรง

💬 ติดตรงไหนถามได้เลย — ท้าทายข้อ 2 (data race) ถ้ารัน -race แล้วไม่เห็น warning ส่งโค้ดมาได้ บางทีต้องออกแบบให้มันชนกันจริง ๆ ถึงจะจับได้
